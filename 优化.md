```javascript
export class TongueSystem {
  constructor(scene) {
    this.scene = scene;
    this.state = 'RETRACTED'; // RETRACTED | FIRING | RETRACTING
    this.progress = 0;
    this.startPos = new THREE.Vector3();
    this.targetPos = new THREE.Vector3();
    this.currentTip = new THREE.Vector3();

      color: 0xf43f5e,
      roughness: 0.2,
      metalness: 0.1
    });

    this.tipGeom = new THREE.SphereGeometry(0.085, 16, 16);

      color: 0xff0055,
      emissive: 0x9f1239,
      roughness: 0.1,
      clearcoat: 1.0
    });
    this.tipMesh = new THREE.Mesh(this.tipGeom, this.tipMat);

    this.group.add(this.shaftMesh, this.tipMesh);
    this.group.visible = false;
    this.scene.add(this.group);
  }

  fire(mouthPos, targetPos) {
    if (this.state !== 'RETRACTED') return;
    this.startPos.copy(mouthPos);
    this.targetPos.copy(targetPos);
    this.currentTip.copy(mouthPos);
    this.progress = 0;
    this.state = 'FIRING';
    this.group.visible = true;
  }

  update(mouthPos, delta) {
    this.startPos.copy(mouthPos);

    if (this.state === 'FIRING') {
      this.progress += delta * 10.0; // 闪电射出
      if (this.progress >= 1.0) {
        this.progress = 1.0;
        this.state = 'RETRACTING';
      }
      this.currentTip.lerpVectors(this.startPos, this.targetPos, this.progress);
    } else if (this.state === 'RETRACTING') {
      this.progress -= delta * 4.5; // 柔韧收回
      if (this.progress <= 0) {
        this.progress = 0;
        this.state = 'RETRACTED';
        this.group.visible = false;
      }
      this.currentTip.lerpVectors(this.startPos, this.targetPos, this.progress);
    } else {
      this.currentTip.copy(mouthPos);
      this.group.visible = false;
    }

    if (this.group.visible) {
      const dist = this.startPos.distanceTo(this.currentTip);
      const mid = new THREE.Vector3().addVectors(this.startPos, this.currentTip).multiplyScalar(0.5);

      this.shaftMesh.position.copy(mid);
      this.shaftMesh.scale.set(1, Math.max(dist, 0.01), 1);
      this.shaftMesh.quaternion.setFromUnitVectors(
        new THREE.Vector3(0, 1, 0),
        new THREE.Vector3().subVectors(this.currentTip, this.startPos).normalize()
      );

      this.tipMesh.position.copy(this.currentTip);
    }
  }

  isActive() {
    return this.state !== 'RETRACTED';
  }
}
```

```javascript
import * as THREE from 'three';

export class LimbIK {
  constructor(spine, shoulderIndex, side = 1, isFront = true) {
    this.spine = spine;
    this.shoulderIndex = shoulderIndex;
    this.side = side;
    this.isFront = isFront;

    this.length1 = 0.32;
    this.length2 = 0.30;
    this.maxReach = (this.length1 + this.length2) * 0.96;

    this.currentFootPos = new THREE.Vector3();
    this.oldFootPos = new THREE.Vector3();
    this.targetFootPos = new THREE.Vector3();
    
    this.isStepping = false;
    this.stepProgress = 0;
    this.stepDuration = 0.16;
    this.stepHeight = 0.25;
    this.footPitch = 0;

    this.groundY = -0.8; // 地面物理高度
    this.form = 'CRAWL'; // 'CRAWL' (陆地爬行) | 'SWIM' (腾云游龙)

    this.currentFootPos.copy(this.getIdealFootPos());
  }

  setForm(form) {
    this.form = form;
    if (form === 'CRAWL') {
      this.currentFootPos.y = this.groundY;
    }
  }

  getShoulderPos() {
    const center = this.spine.getSegment(this.shoulderIndex);
    const forward = this.spine.getSegmentOrientation(this.shoulderIndex);
    const right = new THREE.Vector3(forward.y, -forward.x, 0).normalize().multiplyScalar(this.side * 0.32);
    return new THREE.Vector3().addVectors(center, right);
  }

  getIdealFootPos() {
    const shoulder = this.getShoulderPos();
    const forward = this.spine.getSegmentOrientation(this.shoulderIndex);
    
    if (this.form === 'CRAWL') {
      // 陆地模式：理想落脚点严格投影在地面上
      const lateral = new THREE.Vector3(forward.y, -forward.x, 0).normalize().multiplyScalar(this.side * 0.48);
      const ideal = new THREE.Vector3()
        .addVectors(shoulder, lateral)
        .addScaledVector(forward, this.isFront ? 0.22 : -0.18);
      ideal.y = this.groundY; // 锁定在地面
      return ideal;
    } else {
      // 游龙模式：爪子向后收拢贴身 (流线型)
      return new THREE.Vector3()
        .addVectors(shoulder, new THREE.Vector3(this.side * 0.15, -0.05, -0.25))
        .addScaledVector(forward, -0.3);
    }
  }

  triggerStep() {
    if (this.isStepping || this.form === 'SWIM') return;
    this.isStepping = true;
    this.stepProgress = 0;
    this.oldFootPos.copy(this.currentFootPos);
    this.targetFootPos.copy(this.getIdealFootPos());
  }

  update(delta) {
    if (this.form === 'SWIM') {
      // 游龙模式：四肢平滑向后紧贴身体
      this.currentFootPos.lerp(this.getIdealFootPos(), 0.12);
      this.footPitch = -0.5;
      return;
    }

    // 陆地爬行模式：脚在没迈步前【死死钉在原地】，迈步时走拱形抬腿
    if (this.isStepping) {
      this.stepProgress += delta / this.stepDuration;
      if (this.stepProgress >= 1.0) {
        this.stepProgress = 1.0;
        this.isStepping = false;
        this.currentFootPos.copy(this.targetFootPos);
        this.footPitch = 0;
      } else {
        const t = this.stepProgress;
        this.currentFootPos.lerpVectors(this.oldFootPos, this.targetFootPos, t);
        // 迈步空中起伏
        this.currentFootPos.y = this.groundY + Math.sin(t * Math.PI) * this.stepHeight;
        this.footPitch = Math.sin(t * Math.PI) * 0.5;
      }
    } else {
      this.currentFootPos.y = this.groundY; // 牢牢钉在地面
    }
  }

  solve2BoneIK(root, target) {
    const d = root.distanceTo(target);
    const clampedDist = THREE.MathUtils.clamp(d, 0.05, this.maxReach);

    const cosAlpha = (this.length1 * this.length1 + clampedDist * clampedDist - this.length2 * this.length2) / (2 * this.length1 * clampedDist);
    const alpha = Math.acos(THREE.MathUtils.clamp(cosAlpha, -1, 1));

    const dir = new THREE.Vector3().subVectors(target, root).normalize();
    const bendAxis = this.form === 'SWIM' ? new THREE.Vector3(0, 1, 0) : (this.isFront ? new THREE.Vector3(0, 0, 1) : new THREE.Vector3(0, 0, -1));
    const bendNormal = new THREE.Vector3().crossVectors(bendAxis, dir).multiplyScalar(this.side).normalize();

    const midJoint = new THREE.Vector3()
      .copy(root)
      .addScaledVector(dir, Math.cos(alpha) * this.length1)
      .addScaledVector(bendNormal, Math.sin(alpha) * this.length1);

    const actualFoot = new THREE.Vector3()
      .subVectors(target, midJoint)
      .normalize()
      .multiplyScalar(this.length2)
      .add(midJoint);

    return { root, midJoint, foot: actualFoot, pitch: this.footPitch };
  }
}
```

```javascript
import * as THREE from 'three';

export class SpineChain {
  constructor(numSegments = 20, segmentLength = 0.22) {
    this.numSegments = numSegments;
    this.segmentLength = segmentLength;
    this.joints = [];
    this.waveTimer = 0;
    this.speed = 0;
    this.form = 'CRAWL'; // 'CRAWL' | 'SWIM'

    for (let i = 0; i < numSegments; i++) {
      this.joints.push(new THREE.Vector3(0, -i * segmentLength, 0));
    }
  }

  setForm(form) {
    this.form = form;
  }

  update(headTargetPos, delta = 0.016) {
    const distToTarget = headTargetPos.distanceTo(this.joints[0]);
    this.speed = THREE.MathUtils.lerp(this.speed, distToTarget, 0.15);

    // 头部追踪
    this.joints[0].lerp(headTargetPos, 0.24);

    const isMoving = this.speed > 0.04;
    if (isMoving) {
      this.waveTimer += delta * Math.min(12.0, this.speed * (this.form === 'SWIM' ? 7.0 : 5.0));
    }

    // 骨骼链条约束
    for (let i = 1; i < this.numSegments; i++) {
      const prev = this.joints[i - 1];
      const curr = this.joints[i];

      const dir = new THREE.Vector3().subVectors(curr, prev);

      // 运动时的 S 形游动波
      if (isMoving && i > 1) {
        const waveIntensity = this.form === 'SWIM' ? 0.16 : 0.08;
        const waveAngle = Math.sin(this.waveTimer - i * 0.38) * (this.speed * waveIntensity);
        dir.applyAxisAngle(new THREE.Vector3(0, 0, 1), waveAngle);
      }

      dir.normalize().multiplyScalar(this.segmentLength);
      curr.copy(prev).add(dir);
    }
  }

  getSegment(index) {
    return this.joints[index];
  }

  getSegmentOrientation(index) {
    const p1 = this.joints[Math.max(0, index - 1)];
    const p2 = this.joints[Math.min(this.numSegments - 1, index + 1)];
    return new THREE.Vector3().subVectors(p1, p2).normalize();
  }
}
```
