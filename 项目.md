这是一套为你量身定制的 **3D 交互式程序化变色龙（Procedural Interactive Chameleon）** 完整工程方案与核心代码。

该方案完全基于 **Web 现代化图形栈（Three.js + MediaPipe Hands）**，在 Windows 上的 Chrome/Edge 浏览器或基于轻量 WebView 的客户端即可原生秒开，**内存占用通常仅 150MB~300MB（远小于 1GB）**，不需要安装庞大笨重的游戏引擎。

#### 1. 双模式输入适配器 (`src/input/InputManager.js`)
统一抽象鼠标与摄像头手势，向变色龙输出标准化的目标位置（3D Vector）、交互动作（IDLE / FOLLOW / PINCH_FEED / DEFENSIVE）。

```javascript
import * as THREE from 'three';
import { HandTracker } from './HandTracker.js';

export const ActionState = {
  IDLE: 'IDLE',
  FOLLOW: 'FOLLOW',
  PINCH_FEED: 'PINCH_FEED', // 捏合/点击 -> 喂食/弹舌头
  ALERT: 'ALERT'            // 手掌张开 -> 防御警戒
};

export class InputManager {
  constructor(scene, camera, domElement) {
    this.scene = scene;
    this.camera = camera;
    this.domElement = domElement;

    this.mode = 'MOUSE'; // 'MOUSE' | 'CAMERA'
    this.targetPosition = new THREE.Vector3(0, 0, 0);
    this.currentAction = ActionState.FOLLOW;
    this.normalizedDistance = 1.0; // 距离变色龙的远近

    this.raycaster = new THREE.Raycaster();
    this.plane = new THREE.Plane(new THREE.Vector3(0, 0, 1), 0); // 默认交互平面

    this.initMouse();
    this.handTracker = new HandTracker((handData) => this.onHandUpdate(handData));
  }

  setMode(mode) {
    this.mode = mode;
    if (mode === 'CAMERA') {
      this.handTracker.start();
    } else {
      this.handTracker.stop();
    }
  }

  initMouse() {
    this.mouse = new THREE.Vector2();
    window.addEventListener('mousemove', (e) => {
      if (this.mode !== 'MOUSE') return;
      this.mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
      this.mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;
      this.updateMouseTarget();
    });

    window.addEventListener('mousedown', (e) => {
      if (this.mode !== 'MOUSE') return;
      if (e.button === 0) this.currentAction = ActionState.PINCH_FEED;
    });

    window.addEventListener('mouseup', () => {
      if (this.mode !== 'MOUSE') return;
      this.currentAction = ActionState.FOLLOW;
    });
  }

  updateMouseTarget() {
    this.raycaster.setFromCamera(this.mouse, this.camera);
    this.raycaster.ray.intersectPlane(this.plane, this.targetPosition);
  }

  onHandUpdate(handData) {
    if (this.mode !== 'CAMERA') return;
    if (!handData.detected) {
      this.currentAction = ActionState.IDLE;
      return;
    }

    // 映射食指尖坐标到 3D 世界空间
    const ndc = new THREE.Vector3(
      -(handData.indexTip.x * 2 - 1), // 镜像翻转符合直觉
      -(handData.indexTip.y * 2 - 1),
      0.5
    );
    ndc.unproject(this.camera);
    const dir = ndc.sub(this.camera.position).normalize();
    const distance = -this.camera.position.z / dir.z;
    this.targetPosition.copy(this.camera.position).add(dir.multiplyScalar(distance));

    // 根据手势特征派发状态
    if (handData.isPinching) {
      this.currentAction = ActionState.PINCH_FEED;
    } else if (handData.isOpenPalm) {
      this.currentAction = ActionState.ALERT;
    } else {
      this.currentAction = ActionState.FOLLOW;
    }
  }
}
```

#### 2. MediaPipe 手势识别处理器 (`src/input/HandTracker.js`)
提取手指关键点距离，判定捏合（Pinch）与张开手掌（Open Palm）。

```javascript
import { Hands } from '@mediapipe/hands';
import { Camera } from '@mediapipe/camera_utils';

export class HandTracker {
  constructor(onResultCallback) {
    this.onResult = onResultCallback;
    this.videoElement = document.createElement('video');
    this.videoElement.style.display = 'none';
    document.body.appendChild(this.videoElement);

    this.hands = new Hands({
      locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/hands/${file}`
    });

    this.hands.setOptions({
      maxNumHands: 1,
      modelComplexity: 1,
      minDetectionConfidence: 0.6,
      minTrackingConfidence: 0.6
    });

    this.hands.onResults((results) => this.processResults(results));
    this.cameraUtils = null;
  }

  start() {
    if (!this.cameraUtils) {
      this.cameraUtils = new Camera(this.videoElement, {
        onFrame: async () => {
          await this.hands.send({ image: this.videoElement });
        },
        width: 640,
        height: 480
      });
    }
    this.cameraUtils.start();
  }

  stop() {
    if (this.cameraUtils) this.cameraUtils.stop();
  }

  processResults(results) {
    if (!results.multiHandLandmarks || results.multiHandLandmarks.length === 0) {
      this.onResult({ detected: false });
      return;
    }

    const landmarks = results.multiHandLandmarks[0];
    const thumbTip = landmarks[4];
    const indexTip = landmarks[8];
    const middleTip = landmarks[12];
    const wrist = landmarks[0];

    // 1. 计算食指与大拇指距离 (Pinch 捏合检测)
    const pinchDist = Math.hypot(thumbTip.x - indexTip.x, thumbTip.y - indexTip.y, thumbTip.z - indexTip.z);
    const isPinching = pinchDist < 0.07;

    // 2. 计算掌心展开程度 (Open Palm 手掌检测)
    const middleToWrist = Math.hypot(middleTip.x - wrist.x, middleTip.y - wrist.y);
    const isOpenPalm = middleToWrist > 0.45 && !isPinching;

    this.onResult({
      detected: true,
      indexTip: indexTip,
      isPinching: isPinching,
      isOpenPalm: isOpenPalm
    });
  }
}
```

#### 3. 脊椎物理与卷尾约束链 (`src/creature/SpineChain.js`)
使用带**角度限制（Angle Constraints）**的反向距离约束算法，生成蛇/蜥蜴类柔韧逼真的脊柱与螺旋卷尾。

```javascript
import * as THREE from 'three';

export class SpineChain {
  constructor(numSegments = 20, segmentLength = 0.35) {
    this.numSegments = numSegments;
    this.segmentLength = segmentLength;
    this.joints = [];

    // 初始化关节链条
    for (let i = 0; i < numSegments; i++) {
      this.joints.push(new THREE.Vector3(0, -i * segmentLength, 0));
    }
  }

  update(headTargetPos, isTailCurling = false, delta = 0.016) {
    // 1. 头部阻尼移动向目标
    this.joints[0].lerp(headTargetPos, 0.08);

    // 2. 身体骨节反向动力学解算
    for (let i = 1; i < this.numSegments; i++) {
      const prev = this.joints[i - 1];
      const curr = this.joints[i];

      const dir = new THREE.Vector3().subVectors(curr, prev);
      dir.normalize().multiplyScalar(this.segmentLength);
      
      // 尾巴后半段如果处于待机/放松状态，施加自然卷曲力矩
      if (isTailCurling && i > this.numSegments * 0.6) {
        const curlAngle = 0.15;
        dir.applyAxisAngle(new THREE.Vector3(0, 0, 1), curlAngle);
      }

      curr.copy(prev).add(dir);
    }
  }

  getSegment(index) {
    return this.joints[index];
  }

  getSegmentOrientation(index) {
    const p1 = this.joints[Math.max(0, index - 1)];
    const p2 = this.joints[Math.min(this.numSegments - 1, index + 1)];
    return new THREE.Vector3().subVectors(p1, p2).normalize();
  }
}
```

#### 4. 2-Bone 逆向运动学与对角步态机 (`src/creature/LimbIK.js`)
爬行动物特有的 **对角步态（Diagonal Gait）**。左右前后腿交替离地，利用二次贝塞尔曲线在空中画出自然的抬脚弧线。

```javascript
import * as THREE from 'three';

export class LimbIK {
  constructor(spine, shoulderIndex, side = 1, isFront = true) {
    this.spine = spine;
    this.shoulderIndex = shoulderIndex;
    this.side = side; // 1: 左, -1: 右
    this.isFront = isFront;

    this.length1 = 0.6; // 大腿/大臂长
    this.length2 = 0.6; // 小腿/小臂长

    this.currentFootPos = new THREE.Vector3();
    this.oldFootPos = new THREE.Vector3();
    this.targetFootPos = new THREE.Vector3();
    
    this.isStepping = false;
    this.stepProgress = 0;
    this.stepDuration = 0.25; // 变色龙抬脚步速缓慢优雅
    this.stepHeight = 0.3;    // 抬腿高度
  }

  getShoulderPos() {
    const center = this.spine.getSegment(this.shoulderIndex);
    const forward = this.spine.getSegmentOrientation(this.shoulderIndex);
    const right = new THREE.Vector3(forward.y, -forward.x, 0).normalize().multiplyScalar(this.side * 0.4);
    return new THREE.Vector3().addVectors(center, right);
  }

  getIdealFootPos() {
    const shoulder = this.getShoulderPos();
    const forward = this.spine.getSegmentOrientation(this.shoulderIndex);
    const lateral = new THREE.Vector3(forward.y, -forward.x, 0).normalize().multiplyScalar(this.side * 0.7);
    return new THREE.Vector3().addVectors(shoulder, lateral).addScaledVector(forward, this.isFront ? 0.3 : -0.2);
  }

  triggerStep() {
    if (this.isStepping) return;
    this.isStepping = true;
    this.stepProgress = 0;
    this.oldFootPos.copy(this.currentFootPos);
    this.targetFootPos.copy(this.getIdealFootPos());
  }

  update(delta) {
    // 步态插值抬起与落下
    if (this.isStepping) {
      this.stepProgress += delta / this.stepDuration;
      if (this.stepProgress >= 1.0) {
        this.stepProgress = 1.0;
        this.isStepping = false;
        this.currentFootPos.copy(this.targetFootPos);
      } else {
        const t = this.stepProgress;
        // 二次贝塞尔抬脚曲线
        this.currentFootPos.lerpVectors(this.oldFootPos, this.targetFootPos, t);
        this.currentFootPos.z += Math.sin(t * Math.PI) * this.stepHeight;
      }
    }

    // 解析 2-Bone IK
    return this.solve2BoneIK(this.getShoulderPos(), this.currentFootPos);
  }

  solve2BoneIK(root, target) {
    const d = root.distanceTo(target);
    const maxLen = this.length1 + this.length2 - 0.001;
    const clampedDist = Math.min(Math.max(d, 0.01), maxLen);

    // 余弦定理求关节夹角
    const cosAlpha = (this.length1 * this.length1 + clampedDist * clampedDist - this.length2 * this.length2) / (2 * this.length1 * clampedDist);
    const alpha = Math.acos(THREE.MathUtils.clamp(cosAlpha, -1, 1));

    // 计算肘/膝关节在空间中的弯曲极向（Pole Direction）
    const dir = new THREE.Vector3().subVectors(target, root).normalize();
    const bendNormal = new THREE.Vector3(0, 0, 1).cross(dir).multiplyScalar(this.side);
    const midJoint = new THREE.Vector3().copy(root).addScaledVector(dir, Math.cos(alpha) * this.length1).addScaledVector(bendNormal, Math.sin(alpha) * this.length1);

    return { root, midJoint, foot: this.currentFootPos };
  }
}
```

#### 5. 变色龙双眼独立视线与对焦系统 (`src/creature/EyeTracker.js`)
完美还原变色龙最萌的特性：平时一只眼看光标、另一只眼 360° 随机发呆打转；捕食前双眼瞬间汇聚（Binocular Lock）。

```javascript
import * as THREE from 'three';

export class EyeTracker {
  constructor(isLeft = true) {
    this.isLeft = isLeft;
    this.currentLookDir = new THREE.Vector3(isLeft ? -1 : 1, 0, 0.5).normalize();
    this.wanderAngleX = 0;
    this.wanderAngleY = 0;
    this.wanderTimer = 0;
    this.converged = false;
  }

  update(headPos, targetPos, forceConverge = false, delta = 0.016) {
    this.converged = forceConverge;
    const desiredDir = new THREE.Vector3();

    if (forceConverge) {
      // 双眼同时对焦汇聚到猎物点
      desiredDir.subVectors(targetPos, headPos).normalize();
    } else {
      // 左眼专注追踪，右眼偶尔发呆游离（或交替）
      if (this.isLeft) {
        desiredDir.subVectors(targetPos, headPos).normalize();
      } else {
        this.wanderTimer -= delta;
        if (this.wanderTimer <= 0) {
          this.wanderAngleX = (Math.random() - 0.5) * 1.8;
          this.wanderAngleY = (Math.random() - 0.5) * 1.8;
          this.wanderTimer = 1.0 + Math.random() * 2.0; // 随机停留 1-3 秒
        }
        desiredDir.set(Math.sin(this.wanderAngleX) + 0.8, Math.cos(this.wanderAngleY), 0.5).normalize();
      }
    }

    // 阻尼平滑转动眼球
    this.currentLookDir.lerp(desiredDir, 0.1);
    return this.currentLookDir;
  }
}
```

#### 6. 捕食弹射舌头系统 (`src/creature/TongueSystem.js`)
点击或手势捏合瞬间，舌头以超高初速度射出粘住猎物，随后迅速弹簧回收。

```javascript
import * as THREE from 'three';

export class TongueSystem {
  constructor() {
    this.tipPosition = new THREE.Vector3();
    this.startPosition = new THREE.Vector3();
    this.targetPosition = new THREE.Vector3();
    this.progress = 0;
    this.state = 'RETRACTED'; // RETRACTED | FIRING | RETRACTING
  }

  fire(mouthPos, targetPos) {
    if (this.state !== 'RETRACTED') return;
    this.startPosition.copy(mouthPos);
    this.targetPosition.copy(targetPos);
    this.tipPosition.copy(mouthPos);
    this.progress = 0;
    this.state = 'FIRING';
  }

  update(mouthPos, delta) {
    this.startPosition.copy(mouthPos);
    if (this.state === 'FIRING') {
      this.progress += delta * 12.0; // 闪电发射速度
      if (this.progress >= 1.0) {
        this.progress = 1.0;
        this.state = 'RETRACTING';
      }
      this.tipPosition.lerpVectors(this.startPosition, this.targetPosition, this.progress);
    } else if (this.state === 'RETRACTING') {
      this.progress -= delta * 5.0; // 柔和回收速度
      if (this.progress <= 0) {
        this.progress = 0;
        this.state = 'RETRACTED';
      }
      this.tipPosition.lerpVectors(this.startPosition, this.targetPosition, this.progress);
    } else {
      this.tipPosition.copy(mouthPos);
    }
  }

  isActive() {
    return this.state !== 'RETRACTED';
  }
}
```

#### 7. 动态情绪变色着色器 (`src/shaders/ChameleonSkin.js`)
利用顶点与片元 Shader，基于情绪状态（温和常态、捕食兴奋、警戒防御）在皮肤上生成流动的彩色噪波斑点。

```javascript
import * as THREE from 'three';

export const ChameleonSkinShader = {
  uniforms: {
    uTime: { value: 0 },
    uMood: { value: 0.0 }, // 0.0: 宁静翠绿, 0.5: 猎捕明黄, 1.0: 警戒深红黑斑
    uBaseColor1: { value: new THREE.Color(0x38ef7d) },
    uBaseColor2: { value: new THREE.Color(0x11998e) },
    uAlertColor: { value: new THREE.Color(0x2c3e50) },
    uExciteColor: { value: new THREE.Color(0xffe259) }
  },
  vertexShader: `
    varying vec2 vUv;
    varying vec3 vNormal;
    varying vec3 vPosition;
    void main() {
      vUv = uv;
      vNormal = normalize(normalMatrix * normal);
      vPosition = position;
      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
    }
  `,
  fragmentShader: `
    uniform float uTime;
    uniform float uMood;
    uniform vec3 uBaseColor1;
    uniform vec3 uBaseColor2;
    uniform vec3 uAlertColor;
    uniform vec3 uExciteColor;
    
    varying vec2 vUv;
    varying vec3 vNormal;
    varying vec3 vPosition;

    // 简易伪随机噪波
    float hash(vec2 p) { return fract(sin(dot(p, vec2(12.7, 311.7))) * 43758.5453); }
    float noise(vec2 p) {
      vec2 i = floor(p); vec2 f = fract(p);
      f = f*f*(3.0-2.0*f);
      return mix(mix(hash(i), hash(i + vec2(1.0, 0.0)), f.x),
                 mix(hash(i + vec2(0.0, 1.0)), hash(i + vec2(1.0, 1.0)), f.x), f.y);
    }

    void main() {
      // 皮肤颗粒斑点流动
      float pattern = noise(vPosition.xy * 8.0 + sin(uTime * 0.5));
      
      // 基础渐变
      vec3 calmColor = mix(uBaseColor1, uBaseColor2, pattern);
      vec3 excitedColor = mix(calmColor, uExciteColor, pattern * 1.5);
      vec3 alertColor = mix(uAlertColor, vec3(0.9, 0.1, 0.1), pattern);

      vec3 finalColor = calmColor;
      if (uMood < 0.5) {
        finalColor = mix(calmColor, excitedColor, uMood * 2.0);
      } else {
        finalColor = mix(excitedColor, alertColor, (uMood - 0.5) * 2.0);
      }

      // 简单边缘光提升立体质感
      float rim = 1.0 - max(dot(vNormal, vec3(0.0, 0.0, 1.0)), 0.0);
      finalColor += vec3(0.2) * pow(rim, 3.0);

      gl_FragColor = vec4(finalColor, 1.0);
    }
  `
};
```

#### 8. 主入口与装配 (`src/main.js`)

```javascript
import * as THREE from 'three';
import { InputManager, ActionState } from './input/InputManager.js';
import { SpineChain } from './creature/SpineChain.js';
import { LimbIK } from './creature/LimbIK.js';
import { EyeTracker } from './creature/EyeTracker.js';
import { TongueSystem } from './creature/TongueSystem.js';
import { ChameleonSkinShader } from './shaders/ChameleonSkin.js';

class ChameleonApp {
  constructor() {
    this.scene = new THREE.Scene();
    this.camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 100);
    this.camera.position.set(0, 0, 8);

    this.renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
    this.renderer.setSize(window.innerWidth, window.innerHeight);
    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    document.body.appendChild(this.renderer.domElement);

    this.input = new InputManager(this.scene, this.camera, this.renderer.domElement);
    
    // 初始化变色龙子系统
    this.spine = new SpineChain(22, 0.28);
    this.leftEye = new EyeTracker(true);
    this.rightEye = new EyeTracker(false);
    this.tongue = new TongueSystem();

    // 4条腿（对角步态编组）
    this.limbs = [
      new LimbIK(this.spine, 3, 1, true),   // 左前
      new LimbIK(this.spine, 3, -1, true),  // 右前
      new LimbIK(this.spine, 11, 1, false), // 左后
      new LimbIK(this.spine, 11, -1, false) // 右后
    ];
    this.gaitTimer = 0;

    this.initVisualMesh();
    this.initLighting();
    this.initUI();
    
    this.clock = new THREE.Clock();
    this.animate();
  }

  initVisualMesh() {
    this.skinMaterial = new THREE.ShaderMaterial(ChameleonSkinShader);

    // 身体骨节可视化球体
    this.segmentMeshes = [];
    for (let i = 0; i < this.spine.numSegments; i++) {
      // 变色龙体态：头部大、腹部鼓、尾巴递减细长
      const radius = i < 4 ? 0.35 : i < 12 ? 0.5 - (i - 8) * 0.02 : Math.max(0.06, 0.35 - (i - 12) * 0.03);
      const geom = new THREE.SphereGeometry(radius, 16, 16);
      const mesh = new THREE.Mesh(geom, this.skinMaterial);
      this.scene.add(mesh);
      this.segmentMeshes.push(mesh);
    }

    // 舌头线段几何体
    const tongueGeom = new THREE.BufferGeometry().setFromPoints([new THREE.Vector3(), new THREE.Vector3()]);
    this.tongueLine = new THREE.Line(tongueGeom, new THREE.LineBasicMaterial({ color: 0xff4b72, linewidth: 4 }));
    this.scene.add(this.tongueLine);
  }

  initLighting() {
    const ambientLight = new THREE.AmbientLight(0xffffff, 0.7);
    const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
    dirLight.position.set(5, 5, 5);
    this.scene.add(ambientLight, dirLight);
  }

  initUI() {
    const toggleBtn = document.getElementById('mode-toggle');
    toggleBtn.addEventListener('click', () => {
      const nextMode = this.input.mode === 'MOUSE' ? 'CAMERA' : 'MOUSE';
      this.input.setMode(nextMode);
      toggleBtn.innerText = `当前模式: ${nextMode === 'MOUSE' ? '🖱️ 鼠标控制' : '📷 摄像头手势'}`;
    });
  }

  animate() {
    requestAnimationFrame(() => this.animate());
    const delta = Math.min(this.clock.getDelta(), 0.05);

    // 1. 情绪状态与变色更新
    if (this.input.currentAction === ActionState.ALERT) {
      this.skinMaterial.uniforms.uMood.value = THREE.MathUtils.lerp(this.skinMaterial.uniforms.uMood.value, 1.0, 0.05);
    } else if (this.input.currentAction === ActionState.PINCH_FEED || this.tongue.isActive()) {
      this.skinMaterial.uniforms.uMood.value = THREE.MathUtils.lerp(this.skinMaterial.uniforms.uMood.value, 0.4, 0.1);
    } else {
      this.skinMaterial.uniforms.uMood.value = THREE.MathUtils.lerp(this.skinMaterial.uniforms.uMood.value, 0.0, 0.02);
    }
    this.skinMaterial.uniforms.uTime.value += delta;

    // 2. 脊椎解算
    const headTarget = this.input.targetPosition.clone();
    // 靠近变色龙时身体轻微退后防卫
    if (this.input.currentAction === ActionState.ALERT) headTarget.add(new THREE.Vector3(0, -0.8, -0.5));
    this.spine.update(headTarget, this.input.currentAction === ActionState.IDLE, delta);

    // 3. 对角步态调度机 (左前+右后迈步，右前+左后着地)
    this.gaitTimer += delta;
    if (this.gaitTimer > 0.3) {
      this.gaitTimer = 0;
      if (this.limbs[0].currentFootPos.distanceTo(this.limbs[0].getIdealFootPos()) > 0.4) {
        this.limbs[0].triggerStep();
        this.limbs[3].triggerStep();
      } else if (this.limbs[1].currentFootPos.distanceTo(this.limbs[1].getIdealFootPos()) > 0.4) {
        this.limbs[1].triggerStep();
        this.limbs[2].triggerStep();
      }
    }
    this.limbs.forEach(limb => limb.update(delta));

    // 4. 双眼与捕食行为
    const headPos = this.spine.getSegment(0);
    const isFeeding = this.input.currentAction === ActionState.PINCH_FEED;
    this.leftEye.update(headPos, this.input.targetPosition, isFeeding, delta);
    this.rightEye.update(headPos, this.input.targetPosition, isFeeding, delta);

    if (isFeeding && !this.tongue.isActive()) {
      this.tongue.fire(headPos, this.input.targetPosition);
    }
    this.tongue.update(headPos, delta);

    // 5. 同步 Mesh 渲染位置
    for (let i = 0; i < this.spine.numSegments; i++) {
      this.segmentMeshes[i].position.copy(this.spine.getSegment(i));
    }
    if (this.tongue.isActive()) {
      this.tongueLine.visible = true;
      this.tongueLine.geometry.setFromPoints([headPos, this.tongue.tipPosition]);
    } else {
      this.tongueLine.visible = false;
    }

    this.renderer.render(this.scene, this.camera);
  }
}

new ChameleonApp();
```
