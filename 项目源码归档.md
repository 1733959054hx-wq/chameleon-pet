# 灵动交互水晶灵龙 Pro (Procedural Interactive Chameleon Pet Pro) — 源码归档

> 当前主程序 `src/main.js` 已按最新 `优化.md` 升级为 **2.5D 立体昂首水晶灵龙 + 360° 全景环绕天穹引擎**（Canvas 2D）。三大核心：① 零延迟龙须（刚体伴随平移 + 硬定长绳索约束）；② 2.5D 昂首天鹅颈与远近四肢深度排序（告别趴地壁虎感）；③ 360° 全景星穹 / 傅里叶浮空仙山 / 透视罗盘灵池，支持自动微转与 `A/D`、`←/→`、`Alt+拖拽`、滚轮环视。
> 最新补丁：修复竖直游动时身体"拧麻花/断成几截"——龙身与背鳍改用全场连续法向量（删除逐节 `if (upY > 0)` 翻转），背鳍用平滑统一的 `dorsalSign`，切角改为跨节中心差分。
> 运行方式：`npm install` → `npm run dev`（默认 `http://localhost:5173`）。
> 历史迭代：早期为 Three.js 3D 架构（二阶动力学、1€ Filter、平行传输标架、水晶着色器、PMREM、灵蝶生态与 4 项根治算法），其后为纯 2D 矢量引擎。切换后，`src/creature/`、`src/core/`、`src/shaders/`、`src/world/` 及 `src/input/InputManager.js` **仍保留在磁盘但已不再被主程序引用**，仅作历史源码存档；当前主程序仅依赖 `index.html` 与 `src/input/HandTracker.js`。
> 本归档收录当前工程全部核心源码的完整内容，共 14 个文件：3 个根配置 + 11 个 `src/` 源码（已与本地源码逐字同步）。

```js
class SecondOrder2D {
  constructor(f = 3.2, zeta = 0.88, x0 = 0, y0 = 0) {
    this.x = x0; this.y = y0;
    this.vx = 0; this.vy = 0;
    this.setParams(f, zeta);
  }
  setParams(f, zeta) {
    this.f = Math.max(0.5, f);
    this.zeta = Math.max(0.2, zeta);
    this.k1 = this.zeta / (Math.PI * this.f);
    this.k2 = 1 / ((2 * Math.PI * this.f) ** 2);
  }
  update(tx, ty, dt) {
    if (dt <= 0) return;
    const k2s = Math.max(this.k2, (dt * dt) * 0.5 + dt * this.k1 * 0.5, dt * this.k1);
    this.x += dt * this.vx;
    this.y += dt * this.vy;
    this.vx += dt * (tx - this.x - this.k1 * this.vx) / k2s;
    this.vy += dt * (ty - this.y - this.k1 * this.vy) / k2s;
  }
}

function traceSmoothClosedSpline(ctx, pts) {
  const n = pts.length;
  if (n < 3) return;
  ctx.beginPath();
  ctx.moveTo((pts[n - 1].x + pts[0].x) * 0.5, (pts[n - 1].y + pts[0].y) * 0.5);
  for (let i = 0; i < n; i++) {
    const next = pts[(i + 1) % n];
    ctx.quadraticCurveTo(pts[i].x, pts[i].y, (pts[i].x + next.x) * 0.5, (pts[i].y + next.y) * 0.5);
  }
  ctx.closePath();
}

      footX: s.x, footY: s.y + 18,
      startX: s.x, startY: s.y + 18,
      targetX: s.x, targetY: s.y + 18,
      stepProg: 1.0,
      lift: 0
    };
  }

  spawnButterfly(x, y) {
    if (this.butterflies.length >= 7) this.butterflies.shift();
    this.butterflies.push({
      ox: x, oy: y, x, y,
      phase: Math.random() * Math.PI * 2,
      wing: Math.random() * 10,
      caught: false
    });
    this.emitParticles(x, y, '#38bdf8', 10);
  }

  spawnSwarm() {
    for (let i = 0; i < 3; i++) {
      this.spawnButterfly(
        this.width * (0.28 + Math.random() * 0.55),
        this.height * (0.22 + Math.random() * 0.52)
      );
    }
  }

  emitParticles(x, y, color, count = 12) {
    for (let i = 0; i < count; i++) {
      const a = Math.random() * Math.PI * 2;
      const spd = 35 + Math.random() * 130;
      this.particles.push({
        x, y,
        vx: Math.cos(a) * spd,
        vy: Math.sin(a) * spd,
        life: 1.0,
        size: 2.5 + Math.random() * 3.0,
        color
      });
    }
  }

  initInput() {
    window.addEventListener('contextmenu', (e) => e.preventDefault());

    window.addEventListener('mousemove', (e) => {
      if (this.isPanningSky) {
        const dx = e.clientX - this.panStartX;
        const dy = e.clientY - this.panStartY;
        this.panStartX = e.clientX;
        this.panStartY = e.clientY;
        this.camYaw = wrapAngle(this.camYaw - dx * 0.005);
        this.camPitch = clamp(this.camPitch + dy * 0.003, 0.06, 0.48);
        return;
      }

      if (this.mode !== 'MOUSE') return;
      this.lastInputTime = performance.now();
      const dx = e.clientX - this.pointer.x;
      const dy = e.clientY - this.pointer.y;
      this.pointer.x = e.clientX;
      this.pointer.y = e.clientY;
      const moveSpd = Math.hypot(dx, dy);

      // 抚摸检测
      if (!this.isRightDown && moveSpd > 1.5 && moveSpd < 28) {
        for (let i = 1; i < 16; i++) {
          const seg = this.renderSpine[i];
          if (Math.hypot(this.pointer.x - seg.x, this.pointer.y - seg.y) < this.radii[i] + 18) {
            this.pettingGlow = 1.0;
            this.state = 'PETTING';
            if (Math.random() < 0.25) this.emitParticles(seg.x, seg.y, PALETTES[this.paletteKey].horn, 2);
            return;
          }
        }
      }
    });

    window.addEventListener('mousedown', (e) => {
      if (e.target.closest('#ui-container')) return;
      // 按住中键 或 Alt+左键 可自由拖拽 360° 全景天穹
      if (e.button === 1 || (e.button === 0 && e.altKey)) {
        this.isPanningSky = true;
        this.panStartX = e.clientX;
        this.panStartY = e.clientY;
        return;
      }
      if (this.mode !== 'MOUSE') return;
      this.lastInputTime = performance.now();
      if (e.button === 0) {
        this.triggerTongueStrike();
      } else if (e.button === 2) {
        this.isRightDown = true;
      }
    });

    window.addEventListener('mouseup', (e) => {
      this.isPanningSky = false;
      if (e.button === 2) this.isRightDown = false;
    });

    // 滚轮可调节 2.5D 俯仰视角 (camPitch)
    window.addEventListener('wheel', (e) => {
      this.camPitch = clamp(this.camPitch + e.deltaY * 0.0003, 0.08, 0.48);
    }, { passive: true });

    window.addEventListener('dblclick', (e) => {
      if (e.target.closest('#ui-container')) return;
      this.spawnButterfly(e.clientX, e.clientY);
    });

    window.addEventListener('keydown', (e) => {
      const keys = ['JADE', 'SAKURA', 'CYBER', 'LAVA', 'AURORA'];
      const num = parseInt(e.key);
      if (num >= 1 && num <= 5) this.setPalette(keys[num - 1], num - 1);
      if (e.key === ' ' || e.key === 'Tab') {
        e.preventDefault();
        this.toggleForm();
      }
      if (e.key === 'e' || e.key === 'E') this.isKeyAlert = true;
      if (e.key === 'a' || e.key === 'A' || e.key === 'ArrowLeft') this.keysDown.left = true;
      if (e.key === 'd' || e.key === 'D' || e.key === 'ArrowRight') this.keysDown.right = true;
    });

    window.addEventListener('keyup', (e) => {
      if (e.key === 'e' || e.key === 'E') this.isKeyAlert = false;
      if (e.key === 'a' || e.key === 'A' || e.key === 'ArrowLeft') this.keysDown.left = false;
      if (e.key === 'd' || e.key === 'D' || e.key === 'ArrowRight') this.keysDown.right = false;
    });

    this.handTracker = new HandTracker((handData) => {
      if (this.mode !== 'CAMERA') return;
      if (!handData.detected) {
        this.lastInputTime = 0;
        return;
      }
      this.lastInputTime = performance.now();
      this.pointer.x = (1 - handData.indexTip.x) * this.width;
      this.pointer.y = handData.indexTip.y * this.height;
      this.isRightDown = Boolean(handData.isOpenPalm);
      if (handData.pinchStarted) this.triggerTongueStrike();
    });
  }

  triggerTongueStrike() {
    if (this.tongue.state !== 'IDLE') return;
    const head = this.renderSpine[0];
    let bestPrey = null;
    let bestDist = 350;
    for (const b of this.butterflies) {
      const d = Math.hypot(b.x - head.x, b.y - head.y);
      if (d < bestDist) { bestDist = d; bestPrey = b; }
    }
    this.tongue.state = 'FIRING';
    this.tongue.progress = 0;
    this.tongue.targetPrey = bestPrey;
    this.tongue.tx = bestPrey ? bestPrey.x : this.pointer.x;
    this.tongue.ty = bestPrey ? bestPrey.y : this.pointer.y;
  }

  update(dt) {
    this.time += dt;

    // 1. 更新 360° 全景水平旋转角 camYaw
    if (this.keysDown.left) this.camYawVel = damp(this.camYawVel, -1.1, 8, dt);
    else if (this.keysDown.right) this.camYawVel = damp(this.camYawVel, 1.1, 8, dt);
    else {
      // 根据灵龙水平游动速度自动产生轻微的 360° 全景伴随环绕感
      const autoPan = (this.headDyn.vx / Math.max(600, this.width)) * 0.55;
      this.camYawVel = damp(this.camYawVel, autoPan, 5, dt);
    }
    this.camYaw = wrapAngle(this.camYaw + this.camYawVel * dt);

    // 2. 判定主状态
    const isIdleWander = performance.now() - this.lastInputTime > 5000;
    if (this.tongue.state !== 'IDLE') this.state = 'HUNT';
    else if (this.isRightDown || this.isKeyAlert) this.state = 'ALERT';
    else if (this.pettingGlow > 0.35) this.state = 'PETTING';
    else if (isIdleWander) this.state = 'WANDER';
    else this.state = 'FOLLOW';

    const labels = {
      FOLLOW: 'FOLLOW · 灵珠尾随',
      HUNT: 'HUNT · 弹舌捕食',
      ALERT: 'ALERT · 伏低警戒',
      PETTING: 'PETTING · 愉悦抚摸',
      WANDER: 'WANDER · 360° 巡游'
    };
    const badge = document.getElementById('state-badge');
    if (badge && badge.innerText !== labels[this.state]) badge.innerText = labels[this.state];

    this.alertLevel = damp(this.alertLevel, this.state === 'ALERT' ? 1 : 0, 10, dt);
    this.pettingGlow = damp(this.pettingGlow, 0, 2.8, dt);
    if (this.pulseProgress >= 0) {
      this.pulseProgress += dt * 1.6;
      if (this.pulseProgress > 1.3) this.pulseProgress = -1;
    }

    // 3. 计算连续停靠目标点 (注意：pointer 是屏幕视觉点，需换算回地面基准点供 baseSpine 追踪)
    let targetScreenX = this.pointer.x;
    let targetScreenY = this.pointer.y;
    if (isIdleWander) {
      this.wanderPhase += dt * 0.75;
      this.camYaw = wrapAngle(this.camYaw + dt * 0.12); // 巡游时全景缓慢 360° 环转
      targetScreenX = this.width * 0.52 + Math.sin(this.wanderPhase) * (this.width * 0.26);
      targetScreenY = this.height * 0.52 + Math.sin(this.wanderPhase * 2) * (this.height * 0.16);
    }

    const headLift = this.computePostureElevation(0, this.time);
    const targetGroundX = targetScreenX;
    const targetGroundY = targetScreenY + headLift;

    const head0 = this.baseSpine[0];
    const dxToHead = head0.x - targetGroundX;
    const dyToHead = head0.y - targetGroundY;
    const dist = Math.hypot(dxToHead, dyToHead);
    const followDist = 48;

    let desiredX = head0.x;
    let desiredY = head0.y;
    if (dist > followDist) {
      desiredX = targetGroundX + (dxToHead / dist) * followDist;
      desiredY = targetGroundY + (dyToHead / dist) * followDist;
    }

    const prevGroundHx = head0.x;
    const prevGroundHy = head0.y;
    const prevRenderHx = this.renderSpine[0].x;
    const prevRenderHy = this.renderSpine[0].y;

    this.headDyn.update(desiredX, desiredY, dt);
    head0.x = this.headDyn.x;
    head0.y = this.headDyn.y;

    const instSpeed = Math.hypot(head0.x - prevGroundHx, head0.y - prevGroundHy) / dt;
    this.speed = damp(this.speed, instSpeed < 6 ? 0 : instSpeed, 10, dt);

    // 4. 解算双层脊椎 + 2.5D 离地昂首高度 Z
    const curlFactor = clamp(1 - this.speed / 180, 0, 0.9);
    for (let i = 1; i < this.numSegs; i++) {
      const prev = this.baseSpine[i - 1];
      const curr = this.baseSpine[i];
      const rawAngle = Math.atan2(prev.y - curr.y, prev.x - curr.x);
      let diff = wrapAngle(rawAngle - prev.angle);
      const maxBend = i < 6 ? 0.30 : 0.25;
      diff = clamp(diff, -maxBend, maxBend);

      if (i > 12) {
        const spiralCurl = (i - 12) * 0.042 * curlFactor;
        diff = lerp(diff, spiralCurl, 1 - Math.exp(-5 * dt));
      }

      curr.angle = prev.angle + diff;
      curr.x = prev.x - Math.cos(curr.angle) * this.segLen;
      curr.y = prev.y - Math.sin(curr.angle) * this.segLen;
    }

    this.wavePhase += dt * Math.min(11, this.speed * 0.032);
    const waveAmp = clamp(this.speed * 0.04, 0, 14) * (this.form === 'SWIM' ? 1.45 : 0.75);

    for (let i = 0; i < this.numSegs; i++) {
      const b = this.baseSpine[i];
      const zLift = this.computePostureElevation(i, this.time);
      b.z = zLift;

      const nx = -Math.sin(b.angle), ny = Math.cos(b.angle);
      const env = i === 0 ? 0 : Math.sin((i / (this.numSegs - 1)) * Math.PI * 0.92);
      const wave = Math.sin(this.wavePhase - i * 0.42) * waveAmp * env;

      const gx = b.x + nx * wave;
      const gy = b.y + ny * wave * 0.65;
      this.renderSpine[i].x = gx;
      this.renderSpine[i].groundY = gy;
      // 关键 2.5D 投影：将昂首离地高度 z 映射到屏幕垂直抬升！
      this.renderSpine[i].y = gy - zLift;
      this.renderSpine[i].z = zLift;
    }

    // 重新计算屏幕空间下顺滑的脊椎切角（包含 S 型昂首天鹅颈弧度）
    const rHead = this.renderSpine[0];
    const lookX = this.tongue.state !== 'IDLE' ? this.tongue.tx : targetScreenX;
    const lookY = this.tongue.state !== 'IDLE' ? this.tongue.ty : targetScreenY;
    const desiredHeadAngle = Math.atan2(lookY - rHead.y, lookX - rHead.x);
    rHead.angle = lerpAngle(rHead.angle, desiredHeadAngle, 1 - Math.exp(-15 * dt));

    // ✅ 用前后跨节向量 (prev -> next) 算切线角，消除单节高度差造成的折角突变
    for (let i = 1; i < this.numSegs; i++) {
      const prev = this.renderSpine[i - 1];
      const next = this.renderSpine[Math.min(this.numSegs - 1, i + 1)];
      this.renderSpine[i].angle = Math.atan2(prev.y - next.y, prev.x - next.x);
    }

    // 5. ✅ 零延迟刚体伴随 + 硬定长龙须更新（彻底消灭龙须滞后！）
    const dxRenderHead = rHead.x - prevRenderHx;
    const dyRenderHead = rHead.y - prevRenderHy;
    const whiskerSegLen = 11;
    for (const w of this.whiskers) {
      w.pts[0].x = rHead.x + Math.cos(rHead.angle) * 16 - Math.sin(rHead.angle) * w.side * 8;
      w.pts[0].y = rHead.y + Math.sin(rHead.angle) * 16 + Math.cos(rHead.angle) * w.side * 8;

      for (let i = 1; i < w.pts.length; i++) {
        const prev = w.pts[i - 1], curr = w.pts[i];
        // 刚体伴随平移：直接继承本帧龙头 92% 的位移，0 延迟！
        const advect = 0.92 - i * 0.03;
        curr.x += dxRenderHead * advect;
        curr.y += dyRenderHead * advect;

        const flutter = Math.sin(this.time * 7.5 - i * 0.65 + w.side) * 0.16;
        const idealDir = rHead.angle + Math.PI + w.side * (0.30 + i * 0.07) + flutter;
        const curDir = Math.atan2(curr.y - prev.y, curr.x - prev.x);
        const newDir = lerpAngle(curDir, idealDir, 1 - Math.exp(-(24 - i * 2) * dt));

        // 严格锁死节长，绝不被拉长成橡皮筋
        curr.x = prev.x + Math.cos(newDir) * whiskerSegLen;
        curr.y = prev.y + Math.sin(newDir) * whiskerSegLen;
      }
    }

    // 6. 更新 2.5D 腹下支撑四足步态 (脚踩在 groundY 地面上，撑起悬空的昂首躯干！)
    const groupBusy = [false, false];
    for (const l of this.limbs) if (l.stepProg < 1) groupBusy[l.group] = true;

    for (const l of this.limbs) {
      const anchor = this.renderSpine[l.spineIdx];
      const baseNode = this.baseSpine[l.spineIdx];
      const c = Math.cos(baseNode.angle);

      if (this.form === 'SWIM') {
        // 游龙模式：四肢向后流线型收拢贴腹
        const tuckX = anchor.x - Math.cos(anchor.angle) * 26;
        const tuckY = anchor.y - Math.sin(anchor.angle) * 26 + 10 + l.side * 4;
        l.footX = damp(l.footX, tuckX, 14, dt);
        l.footY = damp(l.footY, tuckY, 14, dt);
        l.stepProg = 1;
        l.lift = 0;
        continue;
      }

      // 2.5D 陆地站立落脚点：位于肩部/胯部正下方的地面 (groundY)，前后错开形成立体透视
      const strideFwd = (l.isFront ? 14 : -10) * c + l.side * 12;
      const depthOffsetY = l.side * 9; // 远侧腿(-1)略高，近侧腿(+1)略低，形成 2.5D 地面透视差
      const idealX = baseNode.x + strideFwd + this.headDyn.vx * 0.065;
      const idealY = anchor.groundY + depthOffsetY + this.headDyn.vy * 0.04;

      if (l.stepProg >= 1) {
        const err = Math.hypot(idealX - l.footX, idealY - l.footY);
        if (err > 28 && (!groupBusy[1 - l.group] || err > 56)) {
          l.stepProg = 0;
          l.startX = l.footX; l.startY = l.footY;
          l.targetX = idealX; l.targetY = idealY;
          groupBusy[l.group] = true;
        }
      } else {
        l.stepProg = Math.min(1, l.stepProg + dt * 7.2);
        const t = l.stepProg;
        const st = t * t * (3 - 2 * t);
        l.footX = lerp(l.startX, idealX, st);
        l.footY = lerp(l.startY, idealY, st);
        l.lift = Math.sin(t * Math.PI) * 14;
      }
    }

    // 7. 弹舌与灵蝶更新
    if (this.tongue.state === 'FIRING') {
      if (this.tongue.targetPrey) {
        this.tongue.tx = this.tongue.targetPrey.x;
        this.tongue.ty = this.tongue.targetPrey.y;
      }
      this.tongue.progress += dt * 9.8;
      if (this.tongue.progress >= 1) {
        this.tongue.progress = 1;
        this.tongue.state = 'RETRACTING';
        if (this.tongue.targetPrey) this.tongue.targetPrey.caught = true;
        this.emitParticles(this.tongue.tx, this.tongue.ty, PALETTES[this.paletteKey].horn, 15);
        this.pulseProgress = 0;
      }
    } else if (this.tongue.state === 'RETRACTING') {
      this.tongue.progress -= dt * 6.0;
      if (this.tongue.progress <= 0) {
        this.tongue.progress = 0;
        this.tongue.state = 'IDLE';
        if (this.tongue.targetPrey) {
          const idx = this.butterflies.indexOf(this.tongue.targetPrey);
          if (idx !== -1) this.butterflies.splice(idx, 1);
          this.tongue.targetPrey = null;
        }
      }
    }

    for (const b of this.butterflies) {
      b.wing += dt * 32;
      if (!b.caught) {
        b.phase += dt * 1.8;
        b.x = b.ox + Math.sin(b.phase * 1.3) * 36;
        b.y = b.oy + Math.cos(b.phase * 1.7) * 24;
      }
    }
    for (let i = this.particles.length - 1; i >= 0; i--) {
      const p = this.particles[i];
      p.life -= dt * 1.9;
      p.x += p.vx * dt;
      p.y += p.vy * dt;
      if (p.life <= 0) this.particles.splice(i, 1);
    }

    this.blinkTimer -= dt;
    if (this.blinkTimer <= 0) {
      this.blinkAmount += dt * 14;
      if (this.blinkAmount >= Math.PI) {
        this.blinkAmount = 0;
        this.blinkTimer = 2.5 + Math.random() * 3.5;
      }
    }
  }

  /**
   * 将 360° 全景方位角 az 投影为屏幕 X 坐标 (视野 FOV = 120°)
   */
  projectAzimuthToX(az) {
    const diff = wrapAngle(az - this.camYaw);
    const fov = (Math.PI * 2) / 3; // 120° 广角全景视野
    return this.width * 0.5 + (diff / (fov * 0.5)) * (this.width * 0.5);
  }

  /**
   * 绘制 360° 全景环绕天穹、浮空水晶仙山、穿云丝带与 3D 透视灵池地面
   */
  draw360PanoramaWorld(ctx, pal) {
    const horizonY = this.height * (0.48 + this.camPitch * 0.35);

    // 1. 天穹与灵池深邃背景渐变
    const skyGrad = ctx.createLinearGradient(0, 0, 0, this.height);
    skyGrad.addColorStop(0, '#040814');
    skyGrad.addColorStop(clamp(horizonY / this.height, 0.2, 0.8), '#0f1f3d');
    skyGrad.addColorStop(1, '#030712');
    ctx.fillStyle = skyGrad;
    ctx.fillRect(0, 0, this.width, this.height);

    // 2. 360° 环绕星穹与极光星带
    ctx.save();
    for (const st of this.panoStars) {
      const sx = this.projectAzimuthToX(st.az);
      if (sx < -20 || sx > this.width + 20) continue;
      const sy = st.el * horizonY * 0.92;
      const alpha = 0.35 + 0.55 * Math.sin(this.time * 2.2 + st.twinkle);
      ctx.fillStyle = `rgba(226, 232, 240, ${alpha.toFixed(2)})`;
      ctx.beginPath();
      ctx.arc(sx, sy, st.size, 0, Math.PI * 2);
      ctx.fill();
    }
    ctx.restore();

    // 3. 360° 周期闭合连绵浮空仙山剪影 (多层傅里叶谐波地形)
    for (let layer = 0; layer < 2; layer++) {
      ctx.save();
      ctx.beginPath();
      ctx.moveTo(0, horizonY);
      const step = 16;
      const fov = (Math.PI * 2) / 3;
      for (let x = 0; x <= this.width + step; x += step) {
        const normX = (x - this.width * 0.5) / (this.width * 0.5);
        const worldAz = this.camYaw + normX * (fov * 0.5) * (layer === 0 ? 0.85 : 1.0);
        const h =
          Math.sin(worldAz * 3 + layer * 2.1) * 28 +
          Math.cos(worldAz * 7 - layer * 1.4) * 16 +
          Math.sin(worldAz * 13) * 7 +
          (layer === 0 ? 54 : 28);
        ctx.lineTo(x, horizonY - Math.max(4, h));
      }
      ctx.lineTo(this.width, horizonY);
      ctx.closePath();
      ctx.fillStyle = layer === 0 ? 'rgba(15, 32, 66, 0.55)' : 'rgba(10, 22, 46, 0.85)';
      ctx.fill();
      ctx.restore();
    }

    // 4. 360° 透视灵池地平线与旋转罗盘经纬线 (赋予地面强烈的 3D 空间纵深)
    ctx.save();
    ctx.strokeStyle = pal.glow;
    ctx.lineWidth = 1;
    ctx.globalAlpha = 0.22;
    const numRays = 24;
    for (let i = 0; i < numRays; i++) {
      const rayAz = (i / numRays) * Math.PI * 2;
      const topX = this.projectAzimuthToX(rayAz);
      if (topX < -this.width * 0.5 || topX > this.width * 1.5) continue;
      const bottomX = this.width * 0.5 + (topX - this.width * 0.5) * 3.6;
      ctx.beginPath();
      ctx.moveTo(topX, horizonY);
      ctx.lineTo(bottomX, this.height);
      ctx.stroke();
    }
    // 水平透视纬线
    for (let j = 1; j <= 6; j++) {
      const t = Math.pow(j / 6, 1.8);
      const y = lerp(horizonY, this.height, t);
      ctx.beginPath();
      ctx.moveTo(0, y);
      ctx.lineTo(this.width, y);
      ctx.stroke();
    }
    ctx.restore();

    // 5. 360° 环绕流云层
    ctx.save();
    for (const c of this.panoClouds) {
      c.az = wrapAngle(c.az + c.speed * 0.016);
      const cx = this.projectAzimuthToX(c.az);
      if (cx < -150 || cx > this.width + 150) continue;
      const cy = c.el * horizonY + Math.sin(this.time + c.az) * 6;
      const grad = ctx.createRadialGradient(cx, cy, 4, cx, cy, c.w);
      grad.addColorStop(0, 'rgba(148, 163, 184, 0.14)');
      grad.addColorStop(1, 'rgba(148, 163, 184, 0)');
      ctx.fillStyle = grad;
      ctx.beginPath();
      ctx.ellipse(cx, cy, c.w, c.h, 0, 0, Math.PI * 2);
      ctx.fill();
    }
    ctx.restore();

    // 6. 右上角精巧的 360° 全景罗盘指示器
    this.drawCompassHUD(ctx, pal);
  }

  draw() {
    const ctx = this.ctx;
    const pal = PALETTES[this.paletteKey];

    // 1. 绘制 360° 全景天穹、仙山与透视灵池
    this.draw360PanoramaWorld(ctx, pal);

    // 2. 绘制引导灵珠
    if (this.state !== 'WANDER') {
      const orbGrad = ctx.createRadialGradient(this.pointer.x, this.pointer.y, 2, this.pointer.x, this.pointer.y, 26);
      orbGrad.addColorStop(0, '#ffffff');
      orbGrad.addColorStop(0.35, pal.horn);
      orbGrad.addColorStop(1, 'rgba(0,0,0,0)');
      ctx.fillStyle = orbGrad;
      ctx.beginPath();
      ctx.arc(this.pointer.x, this.pointer.y, 26, 0, Math.PI * 2);
      ctx.fill();
    }

    // 3. 绘制地面真实投影 (与悬空昂首的龙身纵向分离，塑造强烈 2.5D 站立/悬浮高度感！)
    ctx.save();
    for (let i = 0; i < this.numSegs; i++) {
      const s = this.renderSpine[i];
      const rx = this.radii[i] * 1.25;
      const ry = this.radii[i] * 0.50;
      ctx.fillStyle = 'rgba(2, 6, 23, 0.34)';
      ctx.beginPath();
      ctx.ellipse(s.x, s.groundY + 8, rx, ry, 0, 0, Math.PI * 2);
      ctx.fill();
    }
    ctx.restore();

    // 4. 画家算法第 1 层：绘制【远侧腿 (side = -1)】（位于龙身背面，色调深邃立体）
    this.drawLimbLayer(ctx, pal, -1);

    // 5. 画家算法第 2 层：绘制背部向上挺立的【半透明流光背鳍丝带 + 菱形水晶棘冠 + 尾尖羽扇】
    this.drawUprightDorsalFinAndPlume(ctx, pal);

    // 6. 画家算法第 3 层：绘制弹道舌头
    if (this.tongue.state !== 'IDLE') {
      this.drawBallisticTongue(ctx);
    }

    // 7. 画家算法第 4 层：绘制 2.5D 昂首流线龙身与立体下腹甲
    this.draw25DDragonBody(ctx, pal);

    // 8. 画家算法第 5 层：绘制【近侧腿 (side = +1)】（位于龙身正面，带肩部亮面肌肉与利爪）
    this.drawLimbLayer(ctx, pal, 1);

    // 9. 画家算法第 6 层：绘制 2.5D 昂首立体龙头、零延迟龙须、前后分叉龙角与竖瞳宝石眼
    this.drawSculpted25DDragonHead(ctx, pal);

    // 10. 画家算法第 7 层：绘制发光灵蝶与粒子特效
    this.drawButterfliesAndParticles(ctx);
  }

  /**
   * 分层绘制远侧腿 (-1) 与近侧腿 (+1)：从胸腹向下踩在地面，彻底告别趴地青蛙腿
   */
  drawLimbLayer(ctx, pal, targetSide) {
    for (const l of this.limbs) {
      if (l.side !== targetSide) continue;

      const anchor = this.renderSpine[l.spineIdx];
      // 肩髋关节位于龙身偏腹侧
      const sx = anchor.x + l.side * 4;
      const sy = anchor.y + this.radii[l.spineIdx] * 0.42 + l.side * 3;
      const fx = l.footX;
      const fy = l.footY - l.lift;

      // 2.5D 侧面站立 2-Bone IK：前腿肘关节略向后弯，后腿膝关节略向前弯
      const dx = fx - sx, dy = fy - sy;
      const d = clamp(Math.hypot(dx, dy), 6, (l.L1 + l.L2) * 0.96);
      const baseA = Math.atan2(dy, dx);
      const cosA = (l.L1 ** 2 + d ** 2 - l.L2 ** 2) / (2 * l.L1 * d);
      const alpha = Math.acos(clamp(cosA, -1, 1));
      const facingSign = Math.cos(anchor.angle) >= 0 ? 1 : -1;
      const bendDir = (l.isFront ? -1 : 1) * facingSign;
      const kneeA = baseA + bendDir * alpha;

      const mx = sx + Math.cos(kneeA) * l.L1;
      const my = sy + Math.sin(kneeA) * l.L1;

      ctx.save();
      if (targetSide === -1) {
        // 远侧腿：压暗表现景深遮挡
        ctx.strokeStyle = pal.dark;
        ctx.globalAlpha = 0.85;
      } else {
        const legGrad = ctx.createLinearGradient(sx, sy, fx, fy);
        legGrad.addColorStop(0, pal.body);
        legGrad.addColorStop(1, pal.dark);
        ctx.strokeStyle = legGrad;
      }

      ctx.lineCap = 'round';
      ctx.lineJoin = 'round';

      // 大腿与小腿流线肌肉段
      ctx.lineWidth = 12.5;
      ctx.beginPath(); ctx.moveTo(sx, sy); ctx.lineTo(mx, my); ctx.stroke();

      ctx.lineWidth = 8.5;
      ctx.beginPath(); ctx.moveTo(mx, my); ctx.lineTo(fx, fy); ctx.stroke();

      // 近侧腿肩部高光融合点
      if (targetSide === 1) {
        ctx.fillStyle = pal.belly;
        ctx.globalAlpha = 0.35;
        ctx.beginPath();
        ctx.arc(sx, sy, 6, 0, Math.PI * 2);
        ctx.fill();
        ctx.globalAlpha = 1.0;
      }

      // 水平踩地三趾水晶龙爪
      const fDir = facingSign >= 0 ? 0 : Math.PI;
      ctx.fillStyle = targetSide === 1 ? pal.horn : pal.body;
      for (const offsetA of [-0.32, 0, 0.28]) {
        const ca = fDir + offsetA;
        ctx.beginPath();
        ctx.moveTo(fx, fy - 2);
        ctx.lineTo(fx + Math.cos(ca) * 11, fy + Math.sin(ca) * 4 + 2);
        ctx.lineTo(fx, fy + 2.5);
        ctx.closePath();
        ctx.fill();
      }
      ctx.restore();
    }
  }

  /**
   * 绘制背部向上高高扬起的半透明流光背鳍丝带 + 菱形水晶棘冠 + 尾尖流光羽扇
   */
  drawUprightDorsalFinAndPlume(ctx, pal) {
    const puff = 1 + this.alertLevel * 0.22;
    const ridgeBase = [];
    const finTips = [];

    // ✅ 修复：整条龙共用同一个平滑过渡的背鳍侧向系数，绝不在单节之间跳变！
    const headCos = Math.cos(this.renderSpine[0].angle);
    if (Math.abs(headCos) > 0.15) {
      this.targetDorsalSign = headCos >= 0 ? -1 : 1;
    }
    this.dorsalSign = damp(this.dorsalSign ?? -1, this.targetDorsalSign ?? -1, 8, 0.016);

    for (let i = 1; i < this.numSegs - 1; i++) {
      const s = this.renderSpine[i];
      const t = i / (this.numSegs - 1);
      // 统一的连续背侧法向量
      const upX = -Math.sin(s.angle) * this.dorsalSign;
      const upY =  Math.cos(s.angle) * this.dorsalSign;

      const r = this.radii[i] * puff * 0.78;
      const bx = s.x + upX * r;
      const by = s.y + upY * r;
      ridgeBase.push({ x: bx, y: by });

      const wave = Math.sin(this.time * 6 - i * 0.5) * 3.5;
      const finH = (9 + Math.sin(t * Math.PI) * 15) * (1 + this.alertLevel * 0.4) + wave;
      finTips.push({ x: bx + upX * finH, y: by + upY * finH });
    }

    // 1. 连贯平滑的半透明背鳍膜
    ctx.save();
    ctx.fillStyle = pal.fin;
    ctx.strokeStyle = pal.horn;
    ctx.lineWidth = 1.2;
    ctx.beginPath();
    ctx.moveTo(ridgeBase[0].x, ridgeBase[0].y);
    for (let i = 0; i < finTips.length - 1; i++) {
      const mx = (finTips[i].x + finTips[i + 1].x) * 0.5;
      const my = (finTips[i].y + finTips[i + 1].y) * 0.5;
      ctx.quadraticCurveTo(finTips[i].x, finTips[i].y, mx, my);
    }
    for (let i = ridgeBase.length - 1; i >= 0; i--) {
      ctx.lineTo(ridgeBase[i].x, ridgeBase[i].y);
    }
    ctx.closePath();
    ctx.fill();
    ctx.stroke();

    // 2. 背脊水晶棘刺
    for (let i = 1; i < finTips.length - 1; i += 2) {
      const prevB = ridgeBase[i - 1];
      const nextB = ridgeBase[i + 1];
      const tip = finTips[i];
      const t = i / finTips.length;
      const pulseDist = this.pulseProgress >= 0 ? Math.abs(t - this.pulseProgress) : 99;
      const pulseBoost = Math.exp(-pulseDist * pulseDist * 40);

      ctx.fillStyle = pulseBoost > 0.15 ? '#ffffff' : pal.horn;
      ctx.beginPath();
      ctx.moveTo((prevB.x + ridgeBase[i].x) * 0.5, (prevB.y + ridgeBase[i].y) * 0.5);
      ctx.lineTo(tip.x, tip.y);
      ctx.lineTo((nextB.x + ridgeBase[i].x) * 0.5, (nextB.y + ridgeBase[i].y) * 0.5);
      ctx.closePath();
      ctx.fill();
    }

    // 3. 尾尖流光羽扇
    const tail = this.renderSpine[this.numSegs - 1];
    ctx.translate(tail.x, tail.y);
    ctx.rotate(tail.angle + Math.PI);
    ctx.fillStyle = pal.fin;
    ctx.strokeStyle = pal.belly;
    ctx.beginPath();
    ctx.moveTo(0, 0);
    ctx.quadraticCurveTo(18, -14, 34, -4);
    ctx.quadraticCurveTo(42, 0, 34, 4);
    ctx.quadraticCurveTo(18, 14, 0, 0);
    ctx.fill();
    ctx.stroke();
    ctx.restore();
  }

  /**
   * 绘制 2.5D 昂首流线龙身、立体下腹甲鳞片与流光龙脉
   */
  draw25DDragonBody(ctx, pal) {
    const puff = 1 + this.alertLevel * 0.20;
    const topPts = [];
    const botPts = [];

    for (let i = 0; i < this.numSegs; i++) {
      const s = this.renderSpine[i];
      const r = this.radii[i] * puff;
      // ✅ 修复：使用连续法向量 (-sin, cos)，绝不在循环内做 if (upY > 0) 翻转！
      const nx = -Math.sin(s.angle);
      const ny =  Math.cos(s.angle);

      topPts.push({ x: s.x + nx * r, y: s.y + ny * r });
      botPts.push({ x: s.x - nx * r, y: s.y - ny * r });
    }

    const tail = this.renderSpine[this.numSegs - 1];
    const tip = { x: tail.x - Math.cos(tail.angle) * 9, y: tail.y - Math.sin(tail.angle) * 9 };
    const contour = [...topPts, tip, ...botPts.reverse()];

    ctx.save();
    traceSmoothClosedSpline(ctx, contour);

    const head = this.renderSpine[0];
    const bodyGrad = ctx.createLinearGradient(head.x, head.y - 60, tail.x, tail.y + 60);
    bodyGrad.addColorStop(0, pal.belly);
    bodyGrad.addColorStop(0.4, pal.body);
    bodyGrad.addColorStop(1, pal.dark);
    ctx.fillStyle = bodyGrad;
    ctx.fill();

    // 内部腹甲纹路（沿脊椎中心绘制，任意角度都不翻转）
    ctx.clip();
    ctx.strokeStyle = pal.belly;
    for (let i = 2; i < this.numSegs - 3; i++) {
      const s = this.renderSpine[i];
      const r = this.radii[i] * 0.72;
      ctx.globalAlpha = 0.28;
      ctx.lineWidth = 2.0;
      ctx.beginPath();
      ctx.arc(s.x, s.y, r, s.angle - 1.1, s.angle + 1.1);
      ctx.stroke();

      if (this.pulseProgress >= 0) {
        const t = i / (this.numSegs - 1);
        const d = Math.abs(t - this.pulseProgress);
        const boost = Math.exp(-d * d * 45);
        if (boost > 0.08) {
          ctx.globalAlpha = boost * 0.75;
          ctx.fillStyle = '#ffffff';
          ctx.beginPath();
          ctx.arc(s.x, s.y, r * 0.8, 0, Math.PI * 2);
          ctx.fill();
        }
      }
    }
    ctx.restore();
  }

  /**
   * 绘制 2.5D 侧转透视龙首：含前后分叉水晶龙角、零延迟龙须、腮鳍与宝石竖瞳
   */
  drawSculpted25DDragonHead(ctx, pal) {
    const head = this.renderSpine[0];
    const facingRight = Math.cos(head.angle) >= 0;

    // 1. 绘制零延迟飘逸龙须
    ctx.save();
    ctx.strokeStyle = pal.horn;
    ctx.lineCap = 'round';
    for (const w of this.whiskers) {
      ctx.beginPath();
      ctx.moveTo(w.pts[0].x, w.pts[0].y);
      for (let i = 1; i < w.pts.length - 1; i++) {
        const xc = (w.pts[i].x + w.pts[i + 1].x) * 0.5;
        const yc = (w.pts[i].y + w.pts[i + 1].y) * 0.5;
        ctx.quadraticCurveTo(w.pts[i].x, w.pts[i].y, xc, yc);
      }
      ctx.lineWidth = w.side === 1 ? 2.4 : 1.7;
      ctx.globalAlpha = w.side === 1 ? 0.95 : 0.65;
      ctx.stroke();
    }
    ctx.restore();

    ctx.save();
    ctx.translate(head.x, head.y);
    ctx.rotate(head.angle);
    // 保证无论龙头朝左还是朝右，头顶与龙角永远朝向屏幕上方 (不翻肚皮！)
    if (!facingRight) ctx.scale(1, -1);

    // 2. 远侧龙角 (位于头后方，颜色略深形成 2.5D 立体前后关系)
    ctx.fillStyle = pal.body;
    this.drawBranchingHorn(ctx, -6, -10, 0.85);

    // 3. 2.5D 侧面雕刻感龙颅与龙吻轮廓
    const headGrad = ctx.createLinearGradient(0, -18, 0, 16);
    headGrad.addColorStop(0, pal.belly);
    headGrad.addColorStop(0.45, pal.body);
    headGrad.addColorStop(1, pal.dark);
    ctx.fillStyle = headGrad;
    ctx.beginPath();
    ctx.moveTo(28, 2);                   // 微微上翘的龙吻尖
    ctx.quadraticCurveTo(14, -12, 2, -16); // 饱满额头眉弓
    ctx.quadraticCurveTo(-14, -16, -20, -2); // 后脑
    ctx.quadraticCurveTo(-14, 14, 4, 13);    // 下颌护甲
    ctx.quadraticCurveTo(18, 12, 28, 2);
    ctx.closePath();
    ctx.fill();

    // 4. 脸颊半透明水晶腮鳍
    ctx.fillStyle = pal.belly;
    ctx.globalAlpha = 0.78;
    ctx.beginPath();
    ctx.moveTo(-4, 6);
    ctx.quadraticCurveTo(-20, 20 + this.alertLevel * 8, -32, 12);
    ctx.quadraticCurveTo(-20, 2, -8, 2);
    ctx.closePath();
    ctx.fill();
    ctx.globalAlpha = 1.0;

    // 5. 近侧主水晶分叉龙角 (高亮金芒)
    ctx.fillStyle = pal.horn;
    this.drawBranchingHorn(ctx, -2, -8, 1.05);

    // 6. 灵动宝石竖瞳龙眼 (含眨眼与抚摸眯眼微笑)
    const squint = Math.max(Math.sin(this.blinkAmount), this.pettingGlow * 0.85);
    ctx.save();
    ctx.translate(7, -3.5);
    ctx.scale(1, Math.max(0.12, 1 - squint * 0.85));

    ctx.fillStyle = '#fef08a';
    ctx.beginPath();
    ctx.ellipse(0, 0, 6.2, 4.8, -0.1, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = '#020617';
    ctx.beginPath();
    ctx.ellipse(1.2, 0, 2.2, 4.2, 0, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = '#ffffff';
    ctx.beginPath();
    ctx.arc(2.4, -1.5, 1.5, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();

    ctx.restore();
  }

  drawBranchingHorn(ctx, ox, oy, scale) {
    ctx.save();
    ctx.translate(ox, oy);
    ctx.scale(scale, scale);
    const lift = this.alertLevel * 6;
    // 主角后掠上扬
    ctx.beginPath();
    ctx.moveTo(0, 0);
    ctx.quadraticCurveTo(-18, -14, -40, -22 - lift);
    ctx.quadraticCurveTo(-22, -4, -6, 4);
    ctx.closePath();
    ctx.fill();
    // 副角分叉
    ctx.beginPath();
    ctx.moveTo(-14, -9);
    ctx.lineTo(-26, -24 - lift);
    ctx.lineTo(-20, -7);
    ctx.closePath();
    ctx.fill();
    ctx.restore();
  }

  drawBallisticTongue(ctx) {
    const head = this.renderSpine[0];
    const mouthX = head.x + Math.cos(head.angle) * 24;
    const mouthY = head.y + Math.sin(head.angle) * 24;
    const t = this.tongue.progress;
    const tipX = lerp(mouthX, this.tongue.tx, t);
    const tipY = lerp(mouthY, this.tongue.ty, t);

    if (this.tongue.targetPrey && this.tongue.targetPrey.caught) {
      this.tongue.targetPrey.x = tipX;
      this.tongue.targetPrey.y = tipY;
    }

    ctx.save();
    ctx.strokeStyle = '#fb7185';
    ctx.lineWidth = 5;
    ctx.lineCap = 'round';
    ctx.beginPath();
    ctx.moveTo(mouthX, mouthY);
    const midX = (mouthX + tipX) * 0.5 - Math.sin(head.angle) * Math.sin(t * Math.PI) * 18;
```

## `src/core/MathDynamics.js`
> 【3D 版存档，2.5D 主程序未引用】帧率无关指数阻尼、三维二阶弹簧阻尼、1€ 自适应滤波。

```js
import * as THREE from 'three';

/**
 * 帧率无关的指数平滑阻尼函数 (彻底替代固定系数 lerp)
 * @param {number} current 当前值
 * @param {number} target 目标值
 * @param {number} lambda 衰减速率 (越大越快，建议 5 ~ 25)
 * @param {number} delta 帧时间步长 (秒)
 */
export function damp(current, target, lambda, delta) {
  return THREE.MathUtils.lerp(current, target, 1 - Math.exp(-lambda * delta));
}

export function dampVector3(currentVec, targetVec, lambda, delta) {
  const t = 1 - Math.exp(-lambda * delta);
  currentVec.lerp(targetVec, t);
  return currentVec;
}

/**
 * 三维二阶弹簧阻尼动力学系统 (Second-Order Dynamics 3D)
 * 控制方程: y'' + 4*pi*zeta*f*y' + (2*pi*f)^2 * y = (2*pi*f)^2 * x + 2*pi*f*r*zeta * x'
 */
export class SecondOrderDynamics3D {
  /**
   * @param {number} f 固有频率 (Hz) - 控制响应灵敏度与速度
   * @param {number} zeta 阻尼比 - 控制柔和度与回弹 (0.6~0.8 有生物惯性，>=1 无超调)
   * @param {number} r 初始前馈响应 - >1 预测提速，0 平滑起步，<0 反向蓄力
   * @param {THREE.Vector3} initialPos 初始位置
   */
  constructor(f = 3.2, zeta = 0.72, r = 1.15, initialPos = new THREE.Vector3()) {
    this.xp = initialPos.clone();
    this.y = initialPos.clone();
    this.yd = new THREE.Vector3();
    this. _vx = new THREE.Vector3();
    this.setParameters(f, zeta, r);
  }

  setParameters(f, zeta, r) {
    this.f = Math.max(0.1, f);
    this.zeta = Math.max(0.05, zeta);
    this.r = r;
    this.k1 = this.zeta / (Math.PI * this.f);
    this.k2 = 1 / ((2 * Math.PI * this.f) * (2 * Math.PI * this.f));
    this.k3 = (this.r * this.zeta) / (2 * Math.PI * this.f);
  }

  reset(pos) {
    this.xp.copy(pos);
    this.y.copy(pos);
    this.yd.set(0, 0, 0);
  }

  update(targetPos, delta) {
    if (delta <= 0.00001) return this.y;

    // 估算目标输入速度 x'
    this._vx.subVectors(targetPos, this.xp).divideScalar(delta);
    this.xp.copy(targetPos);

    // 保证任意低帧率下积分绝对稳定的 k2 下界
    const k2Stable = Math.max(
      this.k2,
      (delta * delta) / 2 + (delta * this.k1) / 2,
      delta * this.k1
    );

    // 状态位置更新: y = y + delta * yd
    this.y.addScaledVector(this.yd, delta);

    // 状态加速度更新: yd = yd + delta * (x + k3*vx - y - k1*yd) / k2Stable
    const ax = (targetPos.x + this.k3 * this._vx.x - this.y.x - this.k1 * this.yd.x) / k2Stable;
    const ay = (targetPos.y + this.k3 * this._vx.y - this.y.y - this.k1 * this.yd.y) / k2Stable;
    const az = (targetPos.z + this.k3 * this._vx.z - this.y.z - this.k1 * this.yd.z) / k2Stable;

    this.yd.x += delta * ax;
    this.yd.y += delta * ay;
    this.yd.z += delta * az;

    return this.y;
  }
}

/**
 * 1€ Filter (One-Euro Filter) 自适应低通滤波器
 * 专门用于消除 MediaPipe 摄像头手势在静止时的微颤，同时保持快速挥手时零延迟
 */
class LowPassFilter1D {
  constructor() {
    this.y = null;
  }
  filter(value, alpha) {
    if (this.y === null) {
      this.y = value;
      return value;
    }
    this.y = alpha * value + (1.0 - alpha) * this.y;
    return this.y;
  }
}

export class OneEuroFilter3D {
  constructor(minCutoff = 1.2, beta = 0.04, dCutoff = 1.0) {
    this.minCutoff = minCutoff;
    this.beta = beta;
    this.dCutoff = dCutoff;
    this.xFilter = new LowPassFilter1D();
    this.yFilter = new LowPassFilter1D();
    this.zFilter = new LowPassFilter1D();
    this.dxFilter = new LowPassFilter1D();
    this.dyFilter = new LowPassFilter1D();
    this.dzFilter = new LowPassFilter1D();
    this.lastTime = null;
  }

  alpha(cutoff, dt) {
    const tau = 1.0 / (2.0 * Math.PI * cutoff);
    return 1.0 / (1.0 + tau / dt);
  }

  filter(x, y, z, timestampMs = performance.now()) {
    const dt = this.lastTime === null ? 0.016 : Math.max(0.001, (timestampMs - this.lastTime) / 1000);
    this.lastTime = timestampMs;

    const prevX = this.xFilter.y ?? x;
    const prevY = this.yFilter.y ?? y;
    const prevZ = this.zFilter.y ?? z;

    const dx = (x - prevX) / dt;
    const dy = (y - prevY) / dt;
    const dz = (z - prevZ) / dt;

    const dAlpha = this.alpha(this.dCutoff, dt);
    const edx = this.dxFilter.filter(dx, dAlpha);
    const edy = this.dyFilter.filter(dy, dAlpha);
    const edz = this.dzFilter.filter(dz, dAlpha);

    const cutoffX = this.minCutoff + this.beta * Math.abs(edx);
    const cutoffY = this.minCutoff + this.beta * Math.abs(edy);
    const cutoffZ = this.minCutoff + this.beta * Math.abs(edz);

    return {
      x: this.xFilter.filter(x, this.alpha(cutoffX, dt)),
      y: this.yFilter.filter(y, this.alpha(cutoffY, dt)),
      z: this.zFilter.filter(z, this.alpha(cutoffZ, dt))
    };
  }
}
```

```js
  const frames = curve.computeFrenetFrames(numSegments, false);

  const uvs = [];
  const indices = [];

    const p = points[i];

  update(delta) {
    const dt = Math.max(0.001, delta);
    this.shaderUniforms.uTime.value += dt;

    // 更新脊椎流光脉冲动画
    if (this.shaderUniforms.uPulseProgress.value >= 0.0) {
      this.shaderUniforms.uPulseProgress.value += dt * 1.85;
      if (this.shaderUniforms.uPulseProgress.value > 1.35) {
        this.shaderUniforms.uPulseProgress.value = -1.0;
      }
    }

    const headPos = this.spine.getSegment(0).clone();
    const neckPos = this.spine.getSegment(1);
    const headForward = new THREE.Vector3().subVectors(headPos, neckPos).normalize();

    // 吐舌蓄力：头部后缩蓄势
    if (this.tongue && this.tongue.isWindingUp()) {
      const windup = this.tongue.getWindupProgress();
      headPos.addScaledVector(headForward, -0.075 * Math.sin(windup * Math.PI));
    }

    // 1. 头部定位与朝向
    this.headGroup.position.copy(headPos);
    const headTarget = new THREE.Vector3().addVectors(headPos, headForward);
    this.headGroup.lookAt(headTarget);

    // 2. 龙角弹性阻尼更新
    const headMoveDelta = new THREE.Vector3();
    if (!this.prevHeadInitialized) {
      this.prevHeadPos.copy(headPos);
      this.prevHeadInitialized = true;
    } else {
      headMoveDelta.subVectors(headPos, this.prevHeadPos);
      this.prevHeadPos.copy(headPos);
    }

    const springK = 85.0;
    const damping = 9.0;
    const forceX = -headMoveDelta.x * 26.0 - springK * this.hornFlex.x;
    const forceY = -headMoveDelta.y * 26.0 - springK * this.hornFlex.y;

    this.hornVelocity.x += (forceX - damping * this.hornVelocity.x) * dt;
    this.hornVelocity.y += (forceY - damping * this.hornVelocity.y) * dt;
    this.hornFlex.x += this.hornVelocity.x * dt;
    this.hornFlex.y += this.hornVelocity.y * dt;

    this.hornsGroup.rotation.z = this.hornFlex.x * 0.5;
    this.hornsGroup.rotation.x = this.hornFlex.y * 0.5;

    // 2.5 警戒状态与抚摸状态平滑过渡
    this.alertStrength = damp(this.alertStrength, this.isAlert ? 1.0 : 0.0, 10.0, dt);
    this.pettingStrength = damp(this.pettingStrength, 0.0, 3.5, dt);
    this.shaderUniforms.uAlertLevel.value = this.alertStrength;
    this.shaderUniforms.uPettingGlow.value = this.pettingStrength;

    if (this.alertStrength > 0.001) {
      this.hornsGroup.rotation.x += this.alertStrength * 0.22;
      this.hornsGroup.rotation.z += Math.sin(performance.now() * 0.012) * this.alertStrength * 0.02;
    }

    // 3. ✅ 修复眼球坐标系 Bug：用世界坐标 getWorldPosition 加上世界视线方向进行 lookAt
    const leftEyeWorld = new THREE.Vector3();
    const rightEyeWorld = new THREE.Vector3();
    this.leftEye.group.getWorldPosition(leftEyeWorld);
    this.rightEye.group.getWorldPosition(rightEyeWorld);

    this.leftEye.group.lookAt(leftEyeWorld.add(this.leftEyeTracker.currentLookDir));
    this.rightEye.group.lookAt(rightEyeWorld.add(this.rightEyeTracker.currentLookDir));

    // 抚摸时眼睛轻眯表现享受
    const eyeSquint = 1.0 - this.pettingStrength * 0.45;
    this.leftEye.group.scale.y = eyeSquint;
    this.rightEye.group.scale.y = eyeSquint;

    // 4. 平行传输标架 (Parallel Transport Frame) 构建无扭曲龙身与背棘
    const alertPuff = 1.0 + this.alertStrength * 0.20;
    const posAttr = this.positions;

    let prevTangent = null;
    let prevNormal = new THREE.Vector3(0, 0, 1);
    const spineNormals = [];
    const spineTangents = [];

    for (let i = 0; i < this.bodyJointsCount; i++) {
      const p = this.spine.getSegment(i);
      const prev = this.spine.getSegment(Math.max(0, i - 1));
      const next = this.spine.getSegment(Math.min(this.bodyJointsCount - 1, i + 1));

      const tangent = new THREE.Vector3().subVectors(prev, next);
      if (tangent.lengthSq() < 0.0001) tangent.set(0, 1, 0);
      else tangent.normalize();

      if (prevTangent) {
        const axis = new THREE.Vector3().crossVectors(prevTangent, tangent);
        if (axis.lengthSq() > 0.00001) {
          axis.normalize();
          const angle = Math.acos(THREE.MathUtils.clamp(prevTangent.dot(tangent), -1, 1));
          prevNormal.applyAxisAngle(axis, angle);
        }
      } else {
        const refUp = Math.abs(tangent.z) > 0.9 ? new THREE.Vector3(0, 1, 0) : new THREE.Vector3(0, 0, 1);
        prevNormal.crossVectors(tangent, refUp).normalize();
      }

      const binormal = new THREE.Vector3().crossVectors(tangent, prevNormal).normalize();
      prevNormal.crossVectors(binormal, tangent).normalize();
      prevTangent = tangent.clone();

      spineNormals.push(prevNormal.clone());
      spineTangents.push(tangent.clone());

      const t = i / (this.bodyJointsCount - 1);
      let radius =
        THREE.MathUtils.smoothstep(t, 0.0, 0.18) * 0.24 +
        THREE.MathUtils.smoothstep(t, 0.18, 0.50) * 0.08 -
        THREE.MathUtils.smoothstep(t, 0.50, 1.0) * 0.26;
      radius = Math.max(0.022, radius) * alertPuff;

      for (let j = 0; j <= this.radialSegments; j++) {
        const theta = (j / this.radialSegments) * Math.PI * 2;
        const vx = Math.cos(theta) * radius;
        const vy = Math.sin(theta) * radius;

        const vIdx = (i * (this.radialSegments + 1) + j) * 3;
        posAttr[vIdx]     = p.x + prevNormal.x * vx + binormal.x * vy;
        posAttr[vIdx + 1] = p.y + prevNormal.y * vx + binormal.y * vy;
        posAttr[vIdx + 2] = p.z + prevNormal.z * vx + binormal.z * vy;
      }
    }

    this.bodyGeometry.attributes.position.needsUpdate = true;
    this.bodyGeometry.computeVertexNormals();

    // 4.5 更新背部水晶棘刺位姿 (警戒时棘刺竖立张开)
    for (const item of this.dorsalSpines) {
      const idx = item.jointIndex;
      const p = this.spine.getSegment(idx);
      const t = idx / (this.bodyJointsCount - 1);
      let radius =
        THREE.MathUtils.smoothstep(t, 0.0, 0.18) * 0.24 +
        THREE.MathUtils.smoothstep(t, 0.18, 0.50) * 0.08 -
        THREE.MathUtils.smoothstep(t, 0.50, 1.0) * 0.26;
      radius = Math.max(0.022, radius) * alertPuff;

      const normal = spineNormals[idx];
      const tangent = spineTangents[idx];
      const outward = new THREE.Vector3().crossVectors(tangent, normal).normalize();

      item.mesh.position.copy(p).addScaledVector(outward, radius * 0.9);
      const spineDir = outward.clone().addScaledVector(tangent, -0.35 + this.alertStrength * 0.25).normalize();
      item.mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), spineDir);
      item.mesh.scale.setScalar(item.baseScale * (1.0 + this.alertStrength * 0.35));
    }

    // 5. 四肢骨骼与脚爪解算
    this.limbs.forEach((limb, index) => {
      const { root, midJoint, foot, pitch } = limb.solve2BoneIK(limb.getShoulderPos(), limb.currentFootPos);
      const meshes = this.limbMeshes[index];

      meshes.shoulderSocket.position.copy(root);
      this.orientBone(meshes.upperArm, root, midJoint);
      meshes.elbowJoint.position.copy(midJoint);
      this.orientBone(meshes.lowerArm, midJoint, foot);

      meshes.foot.position.copy(foot);
      meshes.foot.lookAt(new THREE.Vector3(foot.x, foot.y - 1, foot.z + 0.1));
      meshes.foot.rotateX(pitch);
    });

    // 6. 帧率无关的材质颜色平滑过渡
    const colorBlend = 1 - Math.exp(-8.0 * dt);
    this.dragonMaterial.color.lerp(this.targetBodyColor, colorBlend);
    this.hornMaterial.color.lerp(this.targetHornColor, colorBlend);
    this.finMaterial.color.lerp(this.targetHornColor, colorBlend);
    this.shaderUniforms.uBellyColor.value.lerp(this.targetBellyColor, colorBlend);
    this.dragonMaterial.emissive.lerp(this.baseEmissive, colorBlend);
  }

  orientBone(mesh, p1, p2) {
    mesh.position.copy(p1).lerp(p2, 0.5);
    const dir = new THREE.Vector3().subVectors(p2, p1);
    if (dir.lengthSq() > 0.00001) {
      mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), dir.normalize());
    }
  }
```

## `src/creature/SpineChain.js`
> 【3D 版存档，2.5D 主程序未引用】结构链/视觉波双层解耦 + 临界阻尼二阶龙头 + 防穿地。

```js
import * as THREE from 'three';
import { SecondOrderDynamics3D, damp } from '../core/MathDynamics.js';

export class SpineChain {
  constructor(numSegments = 20, segmentLength = 0.22) {
    this.numSegments = numSegments;
    this.segmentLength = segmentLength;
    this.joints = [];        // 表层渲染链：baseJoints + 视觉游动波，仅供网格渲染
    this.baseJoints = [];    // 底层结构链：纯物理定长跟随，无正弦波污染
    this.waveTimer = 0;
    this.speed = 0;
    this.form = 'CRAWL'; // 'CRAWL' | 'SWIM'
    this.alertIntensity = 0;
    this.groundY = -0.8;

    for (let i = 0; i < numSegments; i++) {
      const p = new THREE.Vector3(0, -i * segmentLength * 0.5, -i * 0.05);
      this.joints.push(p.clone());
      this.baseJoints.push(p.clone());
    }

    // 临界阻尼 + 无前馈：f=2.8, zeta=0.95, r=0
    // 取消 r 前馈可杜绝离散采样噪声以冲量形式注入龙头（真凶3）
    this.headDynamics = new SecondOrderDynamics3D(2.8, 0.95, 0.0, this.baseJoints[0]);
  }

  setDynamicsParams(f, zeta) {
    this.headDynamics.setParameters(f, zeta, 0.0);
  }

  setForm(form) {
    this.form = form;
  }

  update(headTargetPos, delta = 0.016) {
    const dt = Math.max(0.001, delta);
    const prevHead = this.baseJoints[0].clone();

    // 1. 更新底层结构链龙头
    this.baseJoints[0].copy(this.headDynamics.update(headTargetPos, dt));
    if (this.form === 'CRAWL') {
      this.baseJoints[0].y = Math.max(this.groundY + 0.22, this.baseJoints[0].y);
    }

    // 真实移动速率 (米/秒)，死区过滤微颤；静止时 speed→0，视觉波完全停摆
    const rawSpeed = this.baseJoints[0].distanceTo(prevHead) / dt;
    this.speed = damp(this.speed, rawSpeed < 0.08 ? 0 : rawSpeed, 8.0, dt);

    // 相位累加器：静止时不推进（0 多动症），移动时平缓推进
    this.waveTimer += dt * Math.min(8.0, this.speed * 2.2);
    const waveAmp = THREE.MathUtils.clamp(this.speed * 0.035, 0, 0.14);

    // 2. 解算纯净的底层结构链 baseJoints（无正弦波污染）
    let prevDir = null;
    for (let i = 1; i < this.numSegments; i++) {
      const prev = this.baseJoints[i - 1];
      const curr = this.baseJoints[i];
      const dir = new THREE.Vector3().subVectors(curr, prev);

      // 地面高度平滑修正放在定长之前：只调整方向，不破坏 segmentLength
      if (this.form === 'CRAWL') {
        const targetFloorY = this.groundY + Math.max(0.05, 0.16 * (1 - i / this.numSegments));
        if (curr.y < targetFloorY) dir.y += (targetFloorY - curr.y) * 0.5;
      }

      // ALERT 伏低：给前胸段方向持续的向下倾向（非坐标硬截断，保持定长无抖动）
      if (this.alertIntensity > 0.001 && i < 10) {
        dir.y -= this.alertIntensity * 0.12 * (1 - i / 10);
      }

      dir.normalize();

      // 限制相邻关节最大折角 (防反折)，共线时跳过避免叉乘零向量
      if (prevDir) {
        const angle = dir.angleTo(prevDir);
        const maxAngle = i < 6 ? 0.35 : 0.25;
        if (angle > maxAngle) {
          const axis = new THREE.Vector3().crossVectors(prevDir, dir);
          if (axis.lengthSq() > 0.0001) dir.copy(prevDir).applyAxisAngle(axis.normalize(), maxAngle);
        }
      }

      curr.copy(prev).addScaledVector(dir, this.segmentLength);
      prevDir = dir.clone();
    }

    // 3. 合成表层渲染链 joints = baseJoints + 横向正弦行波偏移
    this.joints[0].copy(this.baseJoints[0]);
    for (let i = 1; i < this.numSegments; i++) {
      const p = this.baseJoints[i];
      const tangent = new THREE.Vector3().subVectors(this.baseJoints[i - 1], p).normalize();
      // XY 平面内垂直于切线的横向法向量
      const sideNormal = new THREE.Vector3(-tangent.y, tangent.x, 0);
      if (sideNormal.lengthSq() < 0.0001) sideNormal.set(1, 0, 0);
      else sideNormal.normalize();

      // 包络：头颈振幅为 0（头不晃），中后段平滑摆动
      const envelope = Math.sin((i / (this.numSegments - 1)) * Math.PI * 0.85);
      const offset = Math.sin(this.waveTimer - i * 0.45) * waveAmp * envelope;

      this.joints[i].copy(p).addScaledVector(sideNormal, offset);
    }
  }

  getSegment(index) {
    return this.joints[index];
  }

  getSegmentOrientation(index) {
    const p1 = this.baseJoints[Math.max(0, index - 1)];
    const p2 = this.baseJoints[Math.min(this.numSegments - 1, index + 1)];
    const dir = new THREE.Vector3().subVectors(p1, p2);
    return dir.lengthSq() > 0.00001 ? dir.normalize() : new THREE.Vector3(0, 1, 0);
  }
}
```

```js
    this.length2 = 0.32;
    this.maxReach = (this.length1 + this.length2) * 0.96;

    this.currentFootPos = new THREE.Vector3();
    this.oldFootPos = new THREE.Vector3();
    this.targetFootPos = new THREE.Vector3();
    this.prevShoulderPos = new THREE.Vector3();
    this.shoulderVelocity = new THREE.Vector3();
    this.initialized = false;

    this.isStepping = false;
    this.stepProgress = 0;
    this.stepDuration = 0.14;
    this.stepHeight = 0.24;
    this.footPitch = 0;

    this.groundY = -0.8;
    this.form = 'CRAWL';

    this.currentFootPos.copy(this.getIdealFootPos());
  }

  setForm(form) {
    this.form = form;
    if (form === 'CRAWL') {
      this.currentFootPos.y = this.groundY;
    }
  }

  // 宏观身体朝向：用结构链前胸→后腰，完全免疫局部 S 波摆动，脚掌稳如泰山
  getMacroBodyForward() {
    return new THREE.Vector3()
      .subVectors(this.spine.baseJoints[1], this.spine.baseJoints[12])
      .normalize();
  }

  getShoulderPos() {
    const center = this.spine.getSegment(this.shoulderIndex);
    const forward = this.getMacroBodyForward();
    const right = new THREE.Vector3(forward.y, -forward.x, 0);
    if (right.lengthSq() < 0.0001) right.set(1, 0, 0);
    right.normalize().multiplyScalar(this.side * 0.30);
    return new THREE.Vector3().addVectors(center, right);
  }

  getIdealFootPos() {
    const shoulder = this.getShoulderPos();
    const forward = this.getMacroBodyForward();

    if (this.form === 'CRAWL') {
      const lateral = new THREE.Vector3(forward.y, -forward.x, 0);
      if (lateral.lengthSq() < 0.0001) lateral.set(1, 0, 0);
      lateral.normalize().multiplyScalar(this.side * 0.56);

      // 引入速度前瞻 (Velocity Lead): 向移动前方预判落脚点，消除高频碎步
      const leadOffset = this.shoulderVelocity.clone().multiplyScalar(0.085);
      leadOffset.clampLength(0, 0.26);

      const ideal = new THREE.Vector3()
        .addVectors(shoulder, lateral)
        .addScaledVector(forward, this.isFront ? 0.28 : -0.20)
        .add(leadOffset);
      ideal.y = this.groundY;
      return ideal;
    } else {
      // 游龙模式：爪子流线型向后贴拢，并随游动轻微摆动
      return new THREE.Vector3()
        .addVectors(shoulder, new THREE.Vector3(this.side * 0.15, -0.06, -0.22))
        .addScaledVector(forward, -0.32);
    }
  }

  triggerStep() {
    if (this.isStepping || this.form === 'SWIM') return;
    this.isStepping = true;
    this.stepProgress = 0;
    this.oldFootPos.copy(this.currentFootPos);
    this.targetFootPos.copy(this.getIdealFootPos());
  }

  update(delta) {
    const dt = Math.max(0.001, delta);
    const shoulder = this.getShoulderPos();

    if (!this.initialized) {
      this.prevShoulderPos.copy(shoulder);
      this.initialized = true;
    } else {
      const instVel = new THREE.Vector3().subVectors(shoulder, this.prevShoulderPos).divideScalar(dt);
      dampVector3(this.shoulderVelocity, instVel, 14.0, dt);
      this.prevShoulderPos.copy(shoulder);
    }

    if (this.form === 'SWIM') {
      this.isStepping = false;
      dampVector3(this.currentFootPos, this.getIdealFootPos(), 10.0, dt);
      this.footPitch = -0.55;
      return;
    }

    if (this.isStepping) {
      this.stepProgress += dt / this.stepDuration;
      if (this.stepProgress >= 1.0) {
        this.stepProgress = 1.0;
        this.isStepping = false;
        this.currentFootPos.copy(this.targetFootPos);
        this.currentFootPos.y = this.groundY;
        this.footPitch = 0;
      } else {
        const t = this.stepProgress;
        // Smoothstep 曲线使起脚与落脚更轻盈干脆
        const smoothT = t * t * (3 - 2 * t);
        this.currentFootPos.lerpVectors(this.oldFootPos, this.targetFootPos, smoothT);
        this.currentFootPos.y = this.groundY + Math.sin(t * Math.PI) * this.stepHeight;
        this.footPitch = Math.sin(t * Math.PI) * 0.5;
      }
    } else {
      this.currentFootPos.y = this.groundY;
    }
  }

  solve2BoneIK(root, target) {
    const d = root.distanceTo(target);
    const clampedDist = THREE.MathUtils.clamp(d, 0.05, this.maxReach);

    const cosAlpha =
      (this.length1 * this.length1 + clampedDist * clampedDist - this.length2 * this.length2) /
      (2 * this.length1 * clampedDist);
    const alpha = Math.acos(THREE.MathUtils.clamp(cosAlpha, -1, 1));

    const dir = new THREE.Vector3().subVectors(target, root).normalize();
    const bodyForward = this.getMacroBodyForward();
    const bodyRight = new THREE.Vector3(bodyForward.y, -bodyForward.x, 0);
    if (bodyRight.lengthSq() < 0.0001) bodyRight.set(1, 0, 0);
    bodyRight.normalize();

    // 稳定极向量：始终指向身体外侧 + 上方，绝不随 dir.x 变号而 180° 翻转
    const poleVector = new THREE.Vector3()
      .addScaledVector(bodyRight, this.side * 0.7)
      .add(new THREE.Vector3(0, 0.6, 0.3))
      .normalize();

    // Gram-Schmidt 正交化求弯曲法线
    const bendNormal = poleVector.clone().sub(dir.clone().multiplyScalar(poleVector.dot(dir)));
    if (bendNormal.lengthSq() < 0.0001) bendNormal.set(0, 1, 0);
    bendNormal.normalize();

    const midJoint = new THREE.Vector3()
      .copy(root)
      .addScaledVector(dir, Math.cos(alpha) * this.length1)
      .addScaledVector(bendNormal, Math.sin(alpha) * this.length1);

    const actualFoot = new THREE.Vector3()
      .subVectors(target, midJoint)
      .normalize()
      .multiplyScalar(this.length2)
      .add(midJoint);

    return { root, midJoint, foot: actualFoot, pitch: this.footPitch };
  }
}
```

```js
export class TongueSystem {
  constructor(scene) {
    this.scene = scene;

    this.state = 'RETRACTED'; // RETRACTED | WINDUP | FIRING | RETRACTING
    this.progress = 0;
    this.windupDuration = 0.11;
    this.windupProgress = 0;

    this.startPos = new THREE.Vector3();
    this.targetPos = new THREE.Vector3();
    this.currentTip = new THREE.Vector3();
    this.capturedPrey = null;
    this.onCatchCallback = null;

    this.group = new THREE.Group();
    this.group.visible = false;
    this.scene.add(this.group);

    this.shaftMat = new THREE.MeshPhysicalMaterial({
      color: 0xf43f5e,
      roughness: 0.18,
      metalness: 0.05,
      clearcoat: 0.85,
      clearcoatRoughness: 0.08
    });

    this.tipMat = new THREE.MeshPhysicalMaterial({
      color: 0xff1744,
      emissive: 0x8a1232,
      emissiveIntensity: 1.6,
      roughness: 0.1,
      clearcoat: 1.0
    });

    this.tipMesh = new THREE.Mesh(new THREE.SphereGeometry(0.085, 16, 16), this.tipMat);
    this.group.add(this.tipMesh);

    // 预分配舌管几何体缓冲区 (彻底消除每帧 dispose/new TubeGeometry 的 GC 卡顿)
    this.tubularSegments = 20;
    this.radialSegments = 10;
    this.baseRadius = 0.042;
    this.curve = new THREE.CatmullRomCurve3([
      new THREE.Vector3(),
      new THREE.Vector3(),
      new THREE.Vector3(),
      new THREE.Vector3()
    ]);

    this.initReusableTubeGeometry();
  }

  initReusableTubeGeometry() {
    const vertexCount = (this.tubularSegments + 1) * (this.radialSegments + 1);
    this.positions = new Float32Array(vertexCount * 3);
    const indices = [];

    for (let i = 0; i < this.tubularSegments; i++) {
      for (let j = 0; j < this.radialSegments; j++) {
        const a = i * (this.radialSegments + 1) + j;
        const b = (i + 1) * (this.radialSegments + 1) + j;
        const c = (i + 1) * (this.radialSegments + 1) + (j + 1);
        const d = i * (this.radialSegments + 1) + (j + 1);
        indices.push(a, b, d, b, c, d);
      }
    }

    this.tongueGeometry = new THREE.BufferGeometry();
    this.tongueGeometry.setIndex(indices);
    this.tongueGeometry.setAttribute('position', new THREE.BufferAttribute(this.positions, 3));
    this.tongueMesh = new THREE.Mesh(this.tongueGeometry, this.shaftMat);
    this.group.add(this.tongueMesh);
  }

  fire(mouthPos, targetPos, preyEntity = null, onCatch = null) {
    if (this.state !== 'RETRACTED') return;

    this.startPos.copy(mouthPos);
    this.targetPos.copy(targetPos);
    this.capturedPrey = preyEntity;
    this.onCatchCallback = onCatch;

    this.currentTip.copy(mouthPos);
    this.progress = 0;
    this.windupProgress = 0;

    this.state = 'WINDUP';
    this.group.visible = false;
  }

  update(mouthPos, delta) {
    this.startPos.copy(mouthPos);
    if (this.capturedPrey && this.capturedPrey.alive && !this.capturedPrey.isCaught) {
      this.targetPos.copy(this.capturedPrey.position);
    }

    if (this.state === 'WINDUP') {
      this.windupProgress += delta / this.windupDuration;
      if (this.windupProgress >= 1) {
        this.windupProgress = 1;
        this.state = 'FIRING';
        this.group.visible = true;
      }
      return;
    } else if (this.state === 'FIRING') {
      this.progress += delta * 11.0;
      if (this.progress >= 1) {
        this.progress = 1;
        this.state = 'RETRACTING';
        if (this.capturedPrey && this.capturedPrey.alive) {
          this.capturedPrey.isCaught = true;
        }
        if (this.onCatchCallback) {
          this.onCatchCallback(this.targetPos.clone());
        }
      }
    } else if (this.state === 'RETRACTING') {
      this.progress -= delta * 6.2;
      if (this.progress <= 0) {
        this.progress = 0;
        this.state = 'RETRACTED';
        this.group.visible = false;
        if (this.capturedPrey) {
          this.capturedPrey.destroy();
          this.capturedPrey = null;
        }
      }
    }

    if (this.state !== 'RETRACTED') {
      this.updateGeometryInPlace();
    }
  }

  updateGeometryInPlace() {
    const t = this.progress;
    // 弹射采用极速缓出曲线，回收采用弹性缓入
    const easedT = this.state === 'FIRING' ? 1 - Math.pow(1 - t, 3) : t * t * (3 - 2 * t);
    this.currentTip.lerpVectors(this.startPos, this.targetPos, easedT);

    if (this.capturedPrey && this.capturedPrey.isCaught) {
      this.capturedPrey.position.copy(this.currentTip);
    }

    const dir = new THREE.Vector3().subVectors(this.targetPos, this.startPos);
    const totalLength = this.startPos.distanceTo(this.currentTip);
    if (dir.lengthSq() < 0.0001) dir.set(0, 0, 1);
    else dir.normalize();

    const bendAxis = Math.abs(dir.y) > 0.9 ? new THREE.Vector3(1, 0, 0) : new THREE.Vector3(0, 1, 0);
    const side = new THREE.Vector3().crossVectors(dir, bendAxis).normalize();

    const bendAmount = Math.sin(t * Math.PI) * Math.min(0.22, totalLength * 0.32);

    this.curve.points[0].copy(this.startPos);
    this.curve.points[1].lerpVectors(this.startPos, this.currentTip, 0.33).addScaledVector(side, bendAmount * 0.55);
    this.curve.points[2].lerpVectors(this.startPos, this.currentTip, 0.68).addScaledVector(side, -bendAmount);
    this.curve.points[3].copy(this.currentTip);

    const points = this.curve.getPoints(this.tubularSegments);
    const frames = this.curve.computeFrenetFrames(this.tubularSegments, false);
    const posArr = this.positions;

    for (let i = 0; i <= this.tubularSegments; i++) {
      const p = points[i];
      const N = frames.normals[i];
      const B = frames.binormals[i];
      const uTube = i / this.tubularSegments;
      const r = this.baseRadius * (0.75 + 0.35 * uTube);

      for (let j = 0; j <= this.radialSegments; j++) {
        const theta = (j / this.radialSegments) * Math.PI * 2;
        const cos = -Math.cos(theta);
        const sin = Math.sin(theta);

        const idx = (i * (this.radialSegments + 1) + j) * 3;
        posArr[idx]     = p.x + (N.x * cos + B.x * sin) * r;
        posArr[idx + 1] = p.y + (N.y * cos + B.y * sin) * r;
        posArr[idx + 2] = p.z + (N.z * cos + B.z * sin) * r;
      }
    }

    this.tongueGeometry.attributes.position.needsUpdate = true;
    this.tongueGeometry.computeVertexNormals();
    this.tipMesh.position.copy(this.currentTip);
  }

  isActive() {
    return this.state !== 'RETRACTED';
  }

  isWindingUp() {
    return this.state === 'WINDUP';
  }

  getWindupProgress() {
    return this.windupProgress;
  }

  getState() {
    return this.state;
  }
}
```

```js
import * as THREE from 'three';
import { HandTracker } from './HandTracker.js';

export const ActionState = {
  IDLE: 'IDLE',             // 自主巡游呼吸
  FOLLOW: 'FOLLOW',         // 灵珠尾随

      this.currentAction = ActionState.FOLLOW;
    }

  initMouseAndKeys() {

    window.addEventListener('mousemove', (e) => {
      if (this.mode !== 'MOUSE') return;
      this.lastPointerMoveTime = performance.now();
      this.mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
      this.mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;

      prevTarget.copy(this.targetPosition);
      this.updateMouseTarget();
      this.pointerSpeed = this.targetPosition.distanceTo(prevTarget);
    });

    window.addEventListener('mousedown', (e) => {
      if (this.mode !== 'MOUSE') return;
      if (e.target.closest('#ui-container')) return;
      this.lastPointerMoveTime = performance.now();

      if (e.button === 0) {
        this.currentAction = ActionState.PINCH_FEED;
        this.feedTriggered = true;
      } else if (e.button === 2) {
        // ✅ 新增：鼠标右键按住直接触发 ALERT 警戒状态
        this.isRightMouseDown = true;
        this.currentAction = ActionState.ALERT;
      }
    });

    window.addEventListener('mouseup', (e) => {
      if (this.mode !== 'MOUSE') return;
      if (e.button === 2) this.isRightMouseDown = false;
      this.currentAction = this.isRightMouseDown || this.isKeyAlertDown ? ActionState.ALERT : ActionState.FOLLOW;
    });

    window.addEventListener('dblclick', (e) => {
      if (e.target.closest('#ui-container')) return;
      this.mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
      this.mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;
      const pt = new THREE.Vector3();
      this.raycaster.setFromCamera(this.mouse, this.camera);
      this.raycaster.ray.intersectPlane(this.plane, pt);
      this.doubleClickSpawnPos = pt;
    });

    window.addEventListener('keydown', (e) => {
      if (e.key === 'e' || e.key === 'E') {
        this.isKeyAlertDown = true;
        this.currentAction = ActionState.ALERT;
      }
    });

    window.addEventListener('keyup', (e) => {
      if (e.key === 'e' || e.key === 'E') {
        this.isKeyAlertDown = false;
        if (this.mode === 'MOUSE' && !this.isRightMouseDown) {
          this.currentAction = ActionState.FOLLOW;
        }
      }
    });
  }

  updateMouseTarget() {
    this.raycaster.setFromCamera(this.mouse, this.camera);
    this.raycaster.ray.intersectPlane(this.plane, this.targetPosition);
  }

  /**
   * 检测指针是否正在轻抚龙身脊椎段，若是则临时切入 PETTING 愉悦状态
   */
  evaluateBodyInteraction(spine, delta) {
    if (this.mode === 'MOUSE') {
      if (this.isRightMouseDown || this.isKeyAlertDown) {
        this.currentAction = ActionState.ALERT;
        return;
      }
      if (this.currentAction === ActionState.PINCH_FEED) return;

      // 超过 5 秒未移动鼠标则进入自主巡游 IDLE 状态
      if (performance.now() - this.lastPointerMoveTime > 5000) {
        this.currentAction = ActionState.IDLE;
        return;
      }

      // 检测鼠标是否在龙身第 2~14 节附近轻柔滑动 (抚摸)
      if (this.pointerSpeed > 0.004 && this.pointerSpeed < 0.18) {
        for (let i = 2; i < 15; i++) {
          if (this.targetPosition.distanceTo(spine.getSegment(i)) < 0.28) {
            this.currentAction = ActionState.PETTING;
            this.pointerSpeed *= 0.5;
            return;
          }
        }
      }
      this.currentAction = ActionState.FOLLOW;
    }
  }

  onHandUpdate(handData) {
    if (this.mode !== 'CAMERA') return;
    if (!handData.detected) {
      this.currentAction = ActionState.IDLE;
      return;
    }

    const ndc = new THREE.Vector3(
      -(handData.indexTip.x * 2 - 1),
      -(handData.indexTip.y * 2 - 1),
      0.5
    );
    ndc.unproject(this.camera);
    const dir = ndc.sub(this.camera.position).normalize();
    const distance = -this.camera.position.z / dir.z;
    this.targetPosition.copy(this.camera.position).add(dir.multiplyScalar(distance));
    this.targetPosition.z = handData.depthZ || 0;

    if (handData.pinchStarted) {
      this.currentAction = ActionState.PINCH_FEED;
      this.feedTriggered = true;
    } else if (handData.isPinching) {
      this.currentAction = ActionState.PINCH_FEED;
    } else if (handData.isOpenPalm) {
      this.currentAction = ActionState.ALERT;
    } else {
      this.currentAction = ActionState.FOLLOW;
    }
  }
}
```

```js
  }

  processResults(results) {
    if (!results.multiHandLandmarks || results.multiHandLandmarks.length === 0) {
      this.onResult({ detected: false });
      return;
    }

    const lm = results.multiHandLandmarks[0];
    const thumbTip = lm[4];
    const rawIndex = lm[8];
    const middleTip = lm[12];
    const wrist = lm[0];

    // 1€ Filter 自适应滤波
    const smoothedIndex = this.euroFilter.filter(rawIndex.x, rawIndex.y, rawIndex.z, performance.now());

    // 捏合迟滞检测 (Hysteresis: 进入 0.06, 释放 0.085)
    const pinchDist = Math.hypot(
      thumbTip.x - rawIndex.x,
      thumbTip.y - rawIndex.y,
      thumbTip.z - rawIndex.z
    );
    let isPinching = this.isPinchingPrev;
    if (pinchDist < 0.06) isPinching = true;
    else if (pinchDist > 0.085) isPinching = false;

    const pinchStarted = isPinching && !this.isPinchingPrev;
    const pinchReleased = !isPinching && this.isPinchingPrev;
    this.isPinchingPrev = isPinching;

    // 张开手掌检测 + 手部距离摄像头深度估算
    const palmSpan = Math.hypot(middleTip.x - wrist.x, middleTip.y - wrist.y);
    const isOpenPalm = palmSpan > 0.45 && !isPinching;
    const depthOffset = (palmSpan - 0.32) * 3.2; // 将手掌远近映射为 3D Z 轴偏移

    this.onResult({
      detected: true,
      indexTip: smoothedIndex,
      depthZ: Math.max(-1.2, Math.min(1.2, depthOffset)),
      isPinching,
      isOpenPalm,
      pinchStarted,
      pinchReleased
    });
  }
}
```
