# 灵动交互水晶灵龙 (Procedural Interactive Chameleon Pet) — 源码归档

> 一个基于 Three.js + Vite 的程序化生成交互宠物 Demo。支持鼠标 / 摄像头手势（MediaPipe Hands）控制，龙可切换「陆地爬行 / 腾云游龙」两种形态，5 套皮肤，含脊椎链式跟随、四肢 IK、眼球追踪、弯曲弹射舌头捕食、张掌警戒等程序化动画。
>
> 运行方式：`npm install` → `npm run dev`（默认 `http://localhost:5173`）。

## 交互状态机

```
                    ┌──────────────┐
                    │    FOLLOW    │
                    └──────┬───────┘
                           │
                 张掌      │
                           ▼
                    ┌──────────────┐
                    │    ALERT     │
                    │ 抬角/鼓身/聚焦 │
                    └──────┬───────┘
                           │
                     松掌  │
                           ▼
                        FOLLOW


                    捏合开始 (pinchStarted)
                       │
                       ▼
               ┌───────────────┐
               │  PINCH_START  │  ← 只触发一次 feedTriggered
               └───────┬───────┘
                       │
                       ▼
                 舌头弯曲弹射捕食
                       │
                       ▼
                 自动 RETRACT（捏住不放不会连发）
```

```
    │   ├── ChameleonModel.js # 龙的整体模型：头/角/眼/身/四肢网格，材质换肤，龙角弹性阻尼，警戒动画
    │   ├── SpineChain.js     # 脊椎链条：头部跟随目标 + S 形游动波（ALERT 时振幅加强）+ 骨骼约束

        ├── InputManager.js   # 统一输入管理：鼠标 raycast + 手势状态机派发 ActionState（含一次性触发标记）
```

## `src/main.js`
> 应用主类 `DragonApp`：初始化渲染器/光照/地面/灵光诱饵，装配脊椎、四肢、眼球、舌头、模型，绑定 UI 与键盘，并在 `animate()` 主循环中编排所有系统的更新与渲染。 ALERT（张掌）在此写入 `spine.alertIntensity` 与 `dragon.setAlertState()`；吐舌由 `input.feedTriggered` 一次性触发。

```js
import * as THREE from 'three';
import { InputManager, ActionState } from './input/InputManager.js';
import { SpineChain } from './creature/SpineChain.js';
import { LimbIK } from './creature/LimbIK.js';
import { EyeTracker } from './creature/EyeTracker.js';
import { TongueSystem } from './creature/TongueSystem.js';
import { ChameleonModel, PALETTES } from './creature/ChameleonModel.js';

class DragonApp {
  constructor() {
    this.scene = new THREE.Scene();
    this.camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 100);
    this.camera.position.set(0, 0.2, 7.5);

    this.renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
    this.renderer.setSize(window.innerWidth, window.innerHeight);
    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    this.renderer.shadowMap.enabled = true;
    this.renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    this.renderer.toneMapping = THREE.ACESFilmicToneMapping;
    this.renderer.toneMappingExposure = 1.3;
    document.body.appendChild(this.renderer.domElement);

    this.input = new InputManager(this.scene, this.camera, this.renderer.domElement);
    
    this.spine = new SpineChain(20, 0.22);
    this.leftEye = new EyeTracker(true);
    this.rightEye = new EyeTracker(false);
    this.tongue = new TongueSystem(this.scene); // 3D 立体舌头

    this.limbs = [
      new LimbIK(this.spine, 3, 1, true),
      new LimbIK(this.spine, 3, -1, true),
      new LimbIK(this.spine, 10, 1, false),
      new LimbIK(this.spine, 10, -1, false)
    ];

    this.dragon = new ChameleonModel(this.scene, this.spine, this.limbs, this.leftEye, this.rightEye);
    this.currentForm = 'CRAWL'; // 'CRAWL' | 'SWIM'

    this.initLightingAndFloor(); // 删掉黑棍，换上高级阴影地面
    this.initSpiritOrb();
    this.initUI();
    this.initKeyboard();
    
    this.clock = new THREE.Clock();
    window.addEventListener('resize', () => this.onResize());
    this.animate();
  }

  initLightingAndFloor() {
    // 主光 (投射柔和阴影)
    const mainLight = new THREE.DirectionalLight(0xffffff, 2.2);
    mainLight.position.set(4, 7, 5);
    mainLight.castShadow = true;
    mainLight.shadow.mapSize.width = 1024;
    mainLight.shadow.mapSize.height = 1024;
    mainLight.shadow.camera.near = 0.5;
    mainLight.shadow.camera.far = 20;

    // 轮廓光与环境光
    const rimLight = new THREE.DirectionalLight(0x38bdf8, 2.5);
    rimLight.position.set(-6, -2, -4);
    const ambientLight = new THREE.AmbientLight(0x0f172a, 1.4);
    this.scene.add(mainLight, rimLight, ambientLight);

    // ❌ 彻底删除斜向黑棍！
    // ✅ 新增：极简半透明阴影地面，爬行时脚掌牢牢踩在此处
    const floorGeom = new THREE.PlaneGeometry(30, 30);
    const floorMat = new THREE.ShadowMaterial({ opacity: 0.35 });
    this.floor = new THREE.Mesh(floorGeom, floorMat);
    this.floor.rotation.x = -Math.PI * 0.5;
    this.floor.position.y = -0.8;
    this.floor.receiveShadow = true;
    this.scene.add(this.floor);
  }

  initSpiritOrb() {
    // 诱饵小灵光
    const orbGroup = new THREE.Group();
    const core = new THREE.Mesh(new THREE.SphereGeometry(0.1, 16, 16), new THREE.MeshBasicMaterial({ color: 0xffffff }));
    const glow = new THREE.Mesh(new THREE.SphereGeometry(0.18, 16, 16), new THREE.MeshBasicMaterial({ color: 0xfde047, transparent: true, opacity: 0.6 }));
    orbGroup.add(core, glow);
    this.spiritOrb = orbGroup;
    this.scene.add(this.spiritOrb);
  }

  setForm(form) {
    this.currentForm = form;
    this.spine.setForm(form);
    this.limbs.forEach(limb => limb.setForm(form));
    document.getElementById('form-toggle').innerText = `当前形态: ${form === 'CRAWL' ? '🐾 陆地爬行' : '🐉 腾云游龙'}`;
  }

  initUI() {
    // 形态切换按钮
    const formBtn = document.getElementById('form-toggle');
    formBtn.addEventListener('click', () => {
      this.setForm(this.currentForm === 'CRAWL' ? 'SWIM' : 'CRAWL');
    });

    // 模式切换
    const toggleBtn = document.getElementById('mode-toggle');
    toggleBtn.addEventListener('click', () => {
      const nextMode = this.input.mode === 'MOUSE' ? 'CAMERA' : 'MOUSE';
      this.input.setMode(nextMode);
      toggleBtn.innerText = `当前输入: ${nextMode === 'MOUSE' ? '🖱️ 鼠标控制' : '📷 摄像头手势'}`;
    });

    // 调色盘
    const paletteContainer = document.getElementById('palette-container');
    if (paletteContainer) {
      Object.keys(PALETTES).forEach(key => {
        const btn = document.createElement('button');
        btn.className = 'palette-btn';
        btn.style.backgroundColor = '#' + PALETTES[key].body.toString(16).padStart(6, '0');
        btn.title = PALETTES[key].name;
        btn.addEventListener('click', () => this.dragon.setPalette(key));
        paletteContainer.appendChild(btn);
      });
    }
  }

  initKeyboard() {
    const keys = ['JADE', 'SAKURA', 'CYBER', 'LAVA', 'AURORA'];
    window.addEventListener('keydown', (e) => {
      const num = parseInt(e.key);
      if (num >= 1 && num <= 5) this.dragon.setPalette(keys[num - 1]);
      if (e.key === ' ' || e.key === 'Tab') {
        e.preventDefault();
        this.setForm(this.currentForm === 'CRAWL' ? 'SWIM' : 'CRAWL');
      }
    });
  }

  onResize() {
    this.camera.aspect = window.innerWidth / window.innerHeight;
    this.camera.updateProjectionMatrix();
    this.renderer.setSize(window.innerWidth, window.innerHeight);
  }

  animate() {
    requestAnimationFrame(() => this.animate());
    const delta = Math.min(this.clock.getDelta(), 0.04);

    const target = this.input.targetPosition.clone();

    // 爬行模式下，头部保持在地面上方平移
    if (this.currentForm === 'CRAWL') {
      target.y = Math.max(-0.45, target.y);
    }
    this.spiritOrb.position.copy(target);

    // 脊椎解算
    this.spine.update(target, delta);

    // 陆地爬行动力学：脚离开理想位置过远时触发迈步
    if (this.currentForm === 'CRAWL') {
      const stepThreshold = 0.35;
      if (this.limbs[0].currentFootPos.distanceTo(this.limbs[0].getIdealFootPos()) > stepThreshold && !this.limbs[1].isStepping) {
        this.limbs[0].triggerStep();
        this.limbs[3].triggerStep();
      } else if (this.limbs[1].currentFootPos.distanceTo(this.limbs[1].getIdealFootPos()) > stepThreshold && !this.limbs[0].isStepping) {
        this.limbs[1].triggerStep();
        this.limbs[2].triggerStep();
      }
    }
    this.limbs.forEach(limb => limb.update(delta));

    // 双眼注视
    const headPos = this.spine.getSegment(0);
    const action = this.input.currentAction;
    const isFeeding = action === ActionState.PINCH_FEED;
    const isAlert = action === ActionState.ALERT;

    // =========================================================
    // ALERT：张掌 → 龙进入警戒状态
    // =========================================================
    if (isAlert) {
      this.dragon.setAlertState(true);
      this.spine.alertIntensity = THREE.MathUtils.lerp(this.spine.alertIntensity ?? 0, 1, 0.15);
    } else {
      this.dragon.setAlertState(false);
      this.spine.alertIntensity = THREE.MathUtils.lerp(this.spine.alertIntensity ?? 0, 0, 0.12);
    }

    this.leftEye.update(headPos, target, isFeeding, delta);
    this.rightEye.update(headPos, target, isFeeding, delta);

    // 3D 肌肉弹射舌头触发：捏合/点击【瞬间】只触发一次，捏住不放不会连发
    if (isFeeding && this.input.feedTriggered && !this.tongue.isActive()) {
      this.tongue.fire(headPos, target);
      this.dragon.triggerFeedPulse();
      this.input.feedTriggered = false;
    }
    this.tongue.update(headPos, delta);

    // 模型渲染
    this.dragon.update(delta);

    this.renderer.render(this.scene, this.camera);
  }
}

new DragonApp();
```

```js
/**
 * 核心几何算法：生成从粗到极其尖锐的渐变弯曲龙角曲面 (彻底告别平头香烟管)
 */
function createTaperedHornGeometry(curve, numSegments = 28, radialSegments = 14, startRadius = 0.085, endRadius = 0.001) {
  const points = curve.getPoints(numSegments);
  const frames = curve.computeFrenetFrames(numSegments, false);
  
  const positions = [];
  const uvs = [];
  const indices = [];

  for (let i = 0; i <= numSegments; i++) {
    const t = i / numSegments;
    // 渐变衰减曲线：根部粗壮，中段柔顺，末端极速收窄为锋利尖芒
    const radius = THREE.MathUtils.lerp(startRadius, endRadius, Math.pow(t, 0.75));
    const p = points[i];
    const N = frames.normals[i];
    const B = frames.binormals[i];

    for (let j = 0; j <= radialSegments; j++) {
      const u = j / radialSegments;
      const angle = u * Math.PI * 2;
      const sin = Math.sin(angle);
      const cos = -Math.cos(angle);

      const normal = new THREE.Vector3().addScaledVector(N, cos).addScaledVector(B, sin).normalize();
      const vertex = new THREE.Vector3().copy(p).addScaledVector(normal, radius);

      positions.push(vertex.x, vertex.y, vertex.z);
      uvs.push(u, t);
    }
  }

  // 缝合三角形网格
  for (let i = 0; i < numSegments; i++) {
    for (let j = 0; j < radialSegments; j++) {
      const a = i * (radialSegments + 1) + j;
      const b = (i + 1) * (radialSegments + 1) + j;
      const c = (i + 1) * (radialSegments + 1) + (j + 1);
      const d = i * (radialSegments + 1) + (j + 1);
      indices.push(a, b, d, b, c, d);
    }
  }

  const geom = new THREE.BufferGeometry();
  geom.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geom.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  geom.setIndex(indices);
  geom.computeVertexNormals();
  return geom;
}

export class ChameleonModel {
  constructor(scene, spine, limbs, leftEyeTracker, rightEyeTracker) {
    this.scene = scene;
    this.spine = spine;
    this.limbs = limbs;
    this.leftEyeTracker = leftEyeTracker;
    this.rightEyeTracker = rightEyeTracker;

    this.currentPalette = 'JADE';
    this.targetBodyColor = new THREE.Color(PALETTES.JADE.body);
    this.targetHornColor = new THREE.Color(PALETTES.JADE.horn);

    // 弹性动力学参数
    this.hornFlex = new THREE.Vector2(0, 0);
    this.hornVelocity = new THREE.Vector2(0, 0);
    this.prevHeadPos = new THREE.Vector3();
    this.prevHeadInitialized = false; // 防止首帧用 (0,0,0) 计算位移导致龙角弹簧突然抖动

    // 警戒状态（ALERT：张掌触发）
    this.isAlert = false;
    this.alertStrength = 0;

    // 顶级玉石清漆材质
    this.dragonMaterial = new THREE.MeshPhysicalMaterial({
      color: this.targetBodyColor.clone(),
      emissive: new THREE.Color(PALETTES.JADE.emissive),
      roughness: 0.1,
      metalness: 0.05,
      clearcoat: 1.0,
      clearcoatRoughness: 0.05,
      transmission: 0.25
    });

    // 琉璃龙角材质（带有晶莹清漆高光）
    this.hornMaterial = new THREE.MeshPhysicalMaterial({
      color: this.targetHornColor.clone(),
      roughness: 0.05,
      metalness: 0.35,
      clearcoat: 1.0,
      clearcoatRoughness: 0.03
    });

    // 半透明龙耳鳍材质
    this.finMaterial = new THREE.MeshPhysicalMaterial({
      color: this.targetHornColor.clone(),
      roughness: 0.2,
      transmission: 0.75,
      transparent: true,
      opacity: 0.8
    });

    this.jointMaterial = new THREE.MeshStandardMaterial({ color: 0x0f766e, roughness: 0.2 });

    this.initDragonHead();
    this.initContinuousBody();
    this.initAnatomicalLimbs();
  }

  initDragonHead() {
    this.headGroup = new THREE.Group();

    // 1. 饱满神骏的龙颅
    const headGeom = new THREE.SphereGeometry(0.38, 24, 24);
    headGeom.scale(0.85, 0.75, 1.15);
    const headMesh = new THREE.Mesh(headGeom, this.dragonMaterial);
    this.headGroup.add(headMesh);

    // 2. 龙吻与下颌
    const jawGeom = new THREE.ConeGeometry(0.24, 0.45, 16);
    jawGeom.rotateX(Math.PI * 0.5);

  createGemEye(side) {
    const group = new THREE.Group();
    group.position.set(side * 0.26, 0.12, 0.18);

    const base = new THREE.Mesh(new THREE.SphereGeometry(0.14, 16, 16), this.dragonMaterial);

  initContinuousBody() {
    this.radialSegments = 16;
    this.bodyJointsCount = this.spine.numSegments;

    this.bodyGeometry = new THREE.BufferGeometry();
    const vertexCount = this.bodyJointsCount * this.radialSegments;
    this.positions = new Float32Array(vertexCount * 3);

    const indices = [];
    for (let i = 0; i < this.bodyJointsCount - 1; i++) {
      for (let j = 0; j < this.radialSegments; j++) {
        const a = i * this.radialSegments + j;
        const b = (i + 1) * this.radialSegments + j;
        const c = (i + 1) * this.radialSegments + ((j + 1) % this.radialSegments);
        const d = i * this.radialSegments + ((j + 1) % this.radialSegments);
        indices.push(a, b, d, b, c, d);

  update(delta) {
    const headPos = this.spine.getSegment(0);
    const neckPos = this.spine.getSegment(1);

    // 1. 头部朝向
    this.headGroup.position.copy(headPos);
    this.headGroup.lookAt(new THREE.Vector3().subVectors(headPos, neckPos).multiplyScalar(2).add(headPos));

    // 2. 龙角弹性阻尼更新（首帧只做初始化，不产生位移冲量）
    const headMoveDelta = new THREE.Vector3();

    if (!this.prevHeadInitialized) {
      this.prevHeadPos.copy(headPos);
      this.prevHeadInitialized = true;
    } else {
      headMoveDelta.subVectors(headPos, this.prevHeadPos);
      this.prevHeadPos.copy(headPos);
    }

    const springK = 80.0;
    const damping = 8.0;
    const forceX = -headMoveDelta.x * 25.0 - springK * this.hornFlex.x;
    const forceY = -headMoveDelta.y * 25.0 - springK * this.hornFlex.y;

    this.hornVelocity.x += (forceX - damping * this.hornVelocity.x) * delta;
    this.hornVelocity.y += (forceY - damping * this.hornVelocity.y) * delta;
    this.hornFlex.x += this.hornVelocity.x * delta;
    this.hornFlex.y += this.hornVelocity.y * delta;

    this.hornsGroup.rotation.z = this.hornFlex.x * 0.5;
    this.hornsGroup.rotation.x = this.hornFlex.y * 0.5;

    // =========================================================
    // 2.5 警戒状态平滑过渡：抬角 + 鼓身 + 双眼聚焦
    // =========================================================
    const targetAlert = this.isAlert ? 1.0 : 0.0;
    this.alertStrength = THREE.MathUtils.lerp(this.alertStrength, targetAlert, 0.12);

    if (this.alertStrength > 0.001) {
      // 龙角抬起（叠加在弹簧姿态之上，每帧被上面的赋值重置，故不会累积）
      this.hornsGroup.rotation.x += this.alertStrength * 0.08;

      // 双眼快速聚焦到头部正前方
      const alertLookDir = new THREE.Vector3()
        .subVectors(this.spine.getSegment(0), this.spine.getSegment(2))
        .normalize();

      this.leftEyeTracker.currentLookDir.lerp(alertLookDir, this.alertStrength * 0.08);
      this.rightEyeTracker.currentLookDir.lerp(alertLookDir, this.alertStrength * 0.08);
    }

    // 3. 眼神注视
    this.leftEye.group.lookAt(new THREE.Vector3().addVectors(this.leftEye.group.position, this.leftEyeTracker.currentLookDir));
    this.rightEye.group.lookAt(new THREE.Vector3().addVectors(this.rightEye.group.position, this.rightEyeTracker.currentLookDir));

    // 4. 一体化无缝龙身曲面
    // 注：龙身顶点是世界坐标，直接缩放 mesh 会把整条龙平移/拉长，
    //     因此"身体鼓起"用半径膨胀实现，效果等价且不会破坏贴地。
    const alertPuff = 1.0 + this.alertStrength * 0.18;
    const posAttr = this.positions;
    for (let i = 0; i < this.bodyJointsCount; i++) {
      const p = this.spine.getSegment(i);
      const prev = this.spine.getSegment(Math.max(0, i - 1));
      const next = this.spine.getSegment(Math.min(this.bodyJointsCount - 1, i + 1));

      const dir = new THREE.Vector3().subVectors(prev, next).normalize();
      if (dir.lengthSq() < 0.0001) dir.set(0, 1, 0);

      const up = new THREE.Vector3(0, 0, 1);
      const normal = new THREE.Vector3().crossVectors(dir, up).normalize();
      const binormal = new THREE.Vector3().crossVectors(dir, normal).normalize();

      let radius = 0.32;
      if (i < 4) radius = 0.28 + i * 0.02;
      else if (i < 10) radius = 0.35 - (i - 4) * 0.012;
      else radius = THREE.MathUtils.lerp(0.30, 0.02, (i - 10) / (this.bodyJointsCount - 10));

      radius *= alertPuff; // 警戒时身体轻微鼓起

      for (let j = 0; j < this.radialSegments; j++) {
        const theta = (j / this.radialSegments) * Math.PI * 2;
        const vx = Math.cos(theta) * radius;
        const vy = Math.sin(theta) * radius * 1.25;

        const vertex = new THREE.Vector3().copy(p).addScaledVector(normal, vx).addScaledVector(binormal, vy);
        const vIdx = (i * this.radialSegments + j) * 3;
        posAttr[vIdx] = vertex.x;
        posAttr[vIdx + 1] = vertex.y;
        posAttr[vIdx + 2] = vertex.z;
      }
    }

    this.bodyGeometry.setAttribute('position', new THREE.BufferAttribute(posAttr, 3));
    this.bodyGeometry.computeVertexNormals();
    this.bodyGeometry.attributes.position.needsUpdate = true;

    // 5. 四肢骨骼与带俯仰的灵动爪
    this.limbs.forEach((limb, index) => {
      const { root, midJoint, foot, pitch } = limb.solve2BoneIK(limb.getShoulderPos(), limb.currentFootPos);
      const meshes = this.limbMeshes[index];

      meshes.shoulderSocket.position.copy(root);
      this.orientBone(meshes.upperArm, root, midJoint);
      meshes.elbowJoint.position.copy(midJoint);
      this.orientBone(meshes.lowerArm, midJoint, foot);

      meshes.foot.position.copy(foot);
      meshes.foot.lookAt(new THREE.Vector3(foot.x, foot.y - 1, foot.z + 0.1));
      meshes.foot.rotateX(pitch);
    });

    // 6. 材质变色过渡
    this.dragonMaterial.color.lerp(this.targetBodyColor, 0.08);
    this.hornMaterial.color.lerp(this.targetHornColor, 0.08);
    this.finMaterial.color.lerp(this.targetHornColor, 0.08);
  }

  orientBone(mesh, p1, p2) {
    mesh.position.copy(p1).lerp(p2, 0.5);
    mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), new THREE.Vector3().subVectors(p2, p1).normalize());
  }

  setAlertState(active) {
    this.isAlert = active;
  }

  setPalette(paletteKey) {
    if (!PALETTES[paletteKey]) return;
    this.currentPalette = paletteKey;
    const p = PALETTES[paletteKey];
    this.targetBodyColor.setHex(p.body);
    this.targetHornColor.setHex(p.horn);
    this.dragonMaterial.emissive.setHex(p.emissive);
  }

  triggerFeedPulse() {
    this.targetBodyColor.setHex(0xffffff);
    setTimeout(() => {
      this.setPalette(this.currentPalette);
    }, 200);
  }
}
```

## `src/creature/SpineChain.js`
> 脊椎链条系统：以头部目标驱动第 0 节，逐节约束固定段长并叠加 S 形游动波；`alertIntensity`（0~1，由 ALERT 状态写入）会放大波动振幅，表现为紧绷颤抖。提供 `getSegment / getSegmentOrientation` 供身体、四肢、头部引用。

```js
import * as THREE from 'three';

export class SpineChain {
  constructor(numSegments = 20, segmentLength = 0.22) {
    this.numSegments = numSegments;
    this.segmentLength = segmentLength;
    this.joints = [];
    this.waveTimer = 0;
    this.speed = 0;
    this.form = 'CRAWL'; // 'CRAWL' | 'SWIM'
    this.alertIntensity = 0; // 0~1，由外部(ALERT 状态)写入，用于加强脊椎振动

    for (let i = 0; i < numSegments; i++) {
      this.joints.push(new THREE.Vector3(0, -i * segmentLength, 0));
    }
  }

  setForm(form) {
    this.form = form;
  }

  update(headTargetPos, delta = 0.016) {
    const distToTarget = headTargetPos.distanceTo(this.joints[0]);
    this.speed = THREE.MathUtils.lerp(this.speed, distToTarget, 0.15);

    // 头部追踪
    this.joints[0].lerp(headTargetPos, 0.24);

    const isMoving = this.speed > 0.04;
    if (isMoving) {
      this.waveTimer += delta * Math.min(12.0, this.speed * (this.form === 'SWIM' ? 7.0 : 5.0));
    }

    // 骨骼链条约束
    for (let i = 1; i < this.numSegments; i++) {
      const prev = this.joints[i - 1];
      const curr = this.joints[i];

      const dir = new THREE.Vector3().subVectors(curr, prev);

      // 运动时的 S 形游动波（ALERT 时整体振幅加强，表现为紧绷颤抖）
      if (isMoving && i > 1) {
        const alertBoost = 1.0 + this.alertIntensity * 1.2;
        const waveIntensity = (this.form === 'SWIM' ? 0.16 : 0.08) * alertBoost;
        const waveAngle = Math.sin(this.waveTimer - i * 0.38) * (this.speed * waveIntensity);
        dir.applyAxisAngle(new THREE.Vector3(0, 0, 1), waveAngle);
      }

      dir.normalize().multiplyScalar(this.segmentLength);
      curr.copy(prev).add(dir);
    }
  }

  getSegment(index) {
    return this.joints[index];
  }

  getSegmentOrientation(index) {
    const p1 = this.joints[Math.max(0, index - 1)];
    const p2 = this.joints[Math.min(this.numSegments - 1, index + 1)];
    return new THREE.Vector3().subVectors(p1, p2).normalize();
  }
}
```

## `src/creature/LimbIK.js`
> 四肢两段式 IK 系统：根据肩部与当前脚位求解肘部与脚位；陆地模式脚钉死地面、超出阈值触发拱形抬腿迈步，游龙模式四肢平滑收拢贴身；导出 `solve2BoneIK`。

```js
import * as THREE from 'three';

export class LimbIK {
  constructor(spine, shoulderIndex, side = 1, isFront = true) {
    this.spine = spine;
    this.shoulderIndex = shoulderIndex;
    this.side = side;
    this.isFront = isFront;

    this.length1 = 0.32;
    this.length2 = 0.30;
    this.maxReach = (this.length1 + this.length2) * 0.96;

    this.currentFootPos = new THREE.Vector3();
    this.oldFootPos = new THREE.Vector3();
    this.targetFootPos = new THREE.Vector3();
    
    this.isStepping = false;
    this.stepProgress = 0;
    this.stepDuration = 0.16;
    this.stepHeight = 0.25;
    this.footPitch = 0;

    this.groundY = -0.8; // 地面物理高度
    this.form = 'CRAWL'; // 'CRAWL' (陆地爬行) | 'SWIM' (腾云游龙)

    this.currentFootPos.copy(this.getIdealFootPos());
  }

  setForm(form) {
    this.form = form;
    if (form === 'CRAWL') {
      this.currentFootPos.y = this.groundY;
    }
  }

  getShoulderPos() {
    const center = this.spine.getSegment(this.shoulderIndex);
    const forward = this.spine.getSegmentOrientation(this.shoulderIndex);
    const right = new THREE.Vector3(forward.y, -forward.x, 0).normalize().multiplyScalar(this.side * 0.32);
    return new THREE.Vector3().addVectors(center, right);
  }

  getIdealFootPos() {
    const shoulder = this.getShoulderPos();
    const forward = this.spine.getSegmentOrientation(this.shoulderIndex);
    
    if (this.form === 'CRAWL') {
      // 陆地模式：理想落脚点严格投影在地面上
      const lateral = new THREE.Vector3(forward.y, -forward.x, 0).normalize().multiplyScalar(this.side * 0.48);
      const ideal = new THREE.Vector3()
        .addVectors(shoulder, lateral)
        .addScaledVector(forward, this.isFront ? 0.22 : -0.18);
      ideal.y = this.groundY; // 锁定在地面
      return ideal;
    } else {
      // 游龙模式：爪子向后收拢贴身 (流线型)
      return new THREE.Vector3()
        .addVectors(shoulder, new THREE.Vector3(this.side * 0.15, -0.05, -0.25))
        .addScaledVector(forward, -0.3);
    }
  }

  triggerStep() {
    if (this.isStepping || this.form === 'SWIM') return;
    this.isStepping = true;
    this.stepProgress = 0;
    this.oldFootPos.copy(this.currentFootPos);
    this.targetFootPos.copy(this.getIdealFootPos());
  }

  update(delta) {
    if (this.form === 'SWIM') {
      // 游龙模式：四肢平滑向后紧贴身体
      this.currentFootPos.lerp(this.getIdealFootPos(), 0.12);
      this.footPitch = -0.5;
      return;
    }

    // 陆地爬行模式：脚在没迈步前【死死钉在原地】，迈步时走拱形抬腿
    if (this.isStepping) {
      this.stepProgress += delta / this.stepDuration;
      if (this.stepProgress >= 1.0) {
        this.stepProgress = 1.0;
        this.isStepping = false;
        this.currentFootPos.copy(this.targetFootPos);
        this.footPitch = 0;
      } else {
        const t = this.stepProgress;
        this.currentFootPos.lerpVectors(this.oldFootPos, this.targetFootPos, t);
        // 迈步空中起伏
        this.currentFootPos.y = this.groundY + Math.sin(t * Math.PI) * this.stepHeight;
        this.footPitch = Math.sin(t * Math.PI) * 0.5;
      }
    } else {
      this.currentFootPos.y = this.groundY; // 牢牢钉在地面
    }
  }

  solve2BoneIK(root, target) {
    const d = root.distanceTo(target);
    const clampedDist = THREE.MathUtils.clamp(d, 0.05, this.maxReach);

    const cosAlpha = (this.length1 * this.length1 + clampedDist * clampedDist - this.length2 * this.length2) / (2 * this.length1 * clampedDist);
    const alpha = Math.acos(THREE.MathUtils.clamp(cosAlpha, -1, 1));

    const dir = new THREE.Vector3().subVectors(target, root).normalize();
    const bendAxis = this.form === 'SWIM' ? new THREE.Vector3(0, 1, 0) : (this.isFront ? new THREE.Vector3(0, 0, 1) : new THREE.Vector3(0, 0, -1));
    const bendNormal = new THREE.Vector3().crossVectors(bendAxis, dir).multiplyScalar(this.side).normalize();

    const midJoint = new THREE.Vector3()
      .copy(root)
      .addScaledVector(dir, Math.cos(alpha) * this.length1)
      .addScaledVector(bendNormal, Math.sin(alpha) * this.length1);

    const actualFoot = new THREE.Vector3()
      .subVectors(target, midJoint)
      .normalize()
      .multiplyScalar(this.length2)
      .add(midJoint);

    return { root, midJoint, foot: actualFoot, pitch: this.footPitch };
  }
}
```

## `src/creature/EyeTracker.js`
> 眼球追踪系统：根据注视目标计算视线方向；左眼始终锁定目标，右眼有概率进入 `wander` 游离发呆，进食（forceConverge）时双眼强制汇聚到猎物点。ALERT 时的聚焦由 `ChameleonModel` 在眼球更新后叠加。

```js
import * as THREE from 'three';

export class EyeTracker {
  constructor(isLeft = true) {
    this.isLeft = isLeft;
    this.currentLookDir = new THREE.Vector3(isLeft ? -1 : 1, 0, 0.5).normalize();
    this.wanderAngleX = 0;
    this.wanderAngleY = 0;
    this.wanderTimer = 0;
    this.converged = false;
  }

  update(headPos, targetPos, forceConverge = false, delta = 0.016) {
    this.converged = forceConverge;
    const desiredDir = new THREE.Vector3();

    if (forceConverge) {
      // 双眼同时对焦汇聚到猎物点
      desiredDir.subVectors(targetPos, headPos).normalize();
    } else {
      // 左眼专注追踪，右眼偶尔发呆游离（或交替）
      if (this.isLeft) {
        desiredDir.subVectors(targetPos, headPos).normalize();
      } else {
        this.wanderTimer -= delta;
        if (this.wanderTimer <= 0) {
          this.wanderAngleX = (Math.random() - 0.5) * 1.8;
          this.wanderAngleY = (Math.random() - 0.5) * 1.8;
          this.wanderTimer = 1.0 + Math.random() * 2.0; // 随机停留 1-3 秒
        }
        desiredDir.set(Math.sin(this.wanderAngleX) + 0.8, Math.cos(this.wanderAngleY), 0.5).normalize();
      }
    }

    // 阻尼平滑转动眼球
    this.currentLookDir.lerp(desiredDir, 0.1);
    return this.currentLookDir;
  }
}
```

## `src/creature/TongueSystem.js`
> 舌头捕食系统：用 `CatmullRomCurve3` + `TubeGeometry` 生成真正带弧度的 3D 肌肉舌管（不再是直线伸缩圆柱），配发光吸盘；状态机 `RETRACTED → FIRING → RETRACTING`，弯曲量随伸出长度与进度 `sin(t·π)` 动态变化。

```js
import * as THREE from 'three';

export class TongueSystem {
  constructor(scene) {
    this.scene = scene;

    this.state = 'RETRACTED'; // RETRACTED | FIRING | RETRACTING
    this.progress = 0;

    this.startPos = new THREE.Vector3();
    this.targetPos = new THREE.Vector3();
    this.currentTip = new THREE.Vector3();

    this.group = new THREE.Group();
    this.group.visible = false;
    this.scene.add(this.group);

    // =========================================================
    // 舌头材质
    // =========================================================
    this.shaftMat = new THREE.MeshPhysicalMaterial({
      color: 0xf43f5e,
      roughness: 0.18,
      metalness: 0.05,
      clearcoat: 0.8,
      clearcoatRoughness: 0.08
    });

    this.tipMat = new THREE.MeshPhysicalMaterial({
      color: 0xff1744,
      emissive: 0x8a1232,
      emissiveIntensity: 1.5,
      roughness: 0.1,
      clearcoat: 1.0
    });

    // 捕食吸盘
    this.tipMesh = new THREE.Mesh(
      new THREE.SphereGeometry(0.085, 16, 16),
      this.tipMat
    );

    this.group.add(this.tipMesh);

    // 舌头曲线
    this.curve = new THREE.CatmullRomCurve3(
      [
        new THREE.Vector3(),
        new THREE.Vector3(),
        new THREE.Vector3(),
        new THREE.Vector3()
      ],
      false,
      'catmullrom',
      0.5
    );

    this.tongueGeometry = new THREE.TubeGeometry(
      this.curve,
      24,
      0.045,
      10,
      false
    );

    this.tongueMesh = new THREE.Mesh(
      this.tongueGeometry,
      this.shaftMat
    );

    this.group.add(this.tongueMesh);
  }

  fire(mouthPos, targetPos) {
    if (this.state !== 'RETRACTED') return;

    this.startPos.copy(mouthPos);
    this.targetPos.copy(targetPos);

    this.currentTip.copy(mouthPos);
    this.progress = 0;

    this.state = 'FIRING';
    this.group.visible = true;
  }

  update(mouthPos, delta) {
    this.startPos.copy(mouthPos);

    if (this.state === 'FIRING') {
      this.progress += delta * 9.5;

      if (this.progress >= 1) {
        this.progress = 1;
        this.state = 'RETRACTING';
      }
    } else if (this.state === 'RETRACTING') {
      this.progress -= delta * 5.0;

      if (this.progress <= 0) {
        this.progress = 0;
        this.state = 'RETRACTED';
        this.group.visible = false;
      }
    }

    if (this.state !== 'RETRACTED') {
      this.updateGeometry();
    }
  }

  updateGeometry() {
    const t = this.progress;

    this.currentTip.lerpVectors(
      this.startPos,
      this.targetPos,
      t
    );

    // =========================================================
    // 弯曲舌头
    // =========================================================
    const dir = new THREE.Vector3()
      .subVectors(this.targetPos, this.startPos)
      .normalize();

    // 找一个与运动方向尽量正交的方向
    let bendAxis = new THREE.Vector3(0, 1, 0);

    if (Math.abs(dir.dot(bendAxis)) > 0.9) {
      bendAxis.set(1, 0, 0);
    }

    const side = new THREE.Vector3()
      .crossVectors(dir, bendAxis)
      .normalize();

    const up = new THREE.Vector3()
      .crossVectors(side, dir)
      .normalize();

    // 当前真实伸出长度
    const totalLength =
      this.startPos.distanceTo(this.currentTip);

    // 舌头越伸出去，弯曲越明显
    const bendAmount =
      Math.sin(t * Math.PI) *
      Math.min(0.22, totalLength * 0.35);

    const p0 = this.startPos.clone();

    const p1 = new THREE.Vector3().lerpVectors(
      this.startPos,
      this.currentTip,
      0.33
    );

    p1.addScaledVector(
      side,
      bendAmount * 0.55
    );

    const p2 = new THREE.Vector3().lerpVectors(
      this.startPos,
      this.currentTip,
      0.68
    );

    p2.addScaledVector(
      side,
      -bendAmount
    );

    const p3 = this.currentTip.clone();

    // 更新曲线
    this.curve.points = [p0, p1, p2, p3];

    // 重新生成 TubeGeometry
    this.tongueMesh.geometry.dispose();

    this.tongueMesh.geometry =
      new THREE.TubeGeometry(
        this.curve,
        24,
        0.045,
        10,
        false
      );

    // 舌尖
    this.tipMesh.position.copy(this.currentTip);
  }

  isActive() {
    return this.state !== 'RETRACTED';
  }
}
```

## `src/input/InputManager.js`
> 统一输入管理：导出 `ActionState` 动作枚举；封装鼠标 raycast 到交互平面，并把 `HandTracker` 的手势结果映射为 3D 世界目标点 + 动作状态（FOLLOW / PINCH_FEED / ALERT / IDLE）；`feedTriggered` 为一次性触发标记，保证捏合 / 点击只吐舌一次；支持 MOUSE / CAMERA 模式切换。

```js
import * as THREE from 'three';
import { HandTracker } from './HandTracker.js';

export const ActionState = {
  IDLE: 'IDLE',
  FOLLOW: 'FOLLOW',
  PINCH_FEED: 'PINCH_FEED', // 捏合/点击 -> 喂食/弹舌头
  ALERT: 'ALERT'            // 手掌张开 -> 防御警戒
};

export class InputManager {
  constructor(scene, camera, domElement) {
    this.scene = scene;
    this.camera = camera;
    this.domElement = domElement;

    this.mode = 'MOUSE'; // 'MOUSE' | 'CAMERA'
    this.targetPosition = new THREE.Vector3(0, 0, 0);
    this.currentAction = ActionState.FOLLOW;
    this.normalizedDistance = 1.0; // 距离变色龙的远近
    this.feedTriggered = false;    // 一次性触发标记：捏合/按下瞬间置位，被主循环消费后清零

    this.raycaster = new THREE.Raycaster();
    this.plane = new THREE.Plane(new THREE.Vector3(0, 0, 1), 0); // 默认交互平面

    this.initMouse();
    this.handTracker = new HandTracker((handData) => this.onHandUpdate(handData));
  }

  setMode(mode) {
    this.mode = mode;
    if (mode === 'CAMERA') {
      this.handTracker.start();
    } else {
      this.handTracker.stop();
    }
  }

  initMouse() {
    this.mouse = new THREE.Vector2();
    window.addEventListener('mousemove', (e) => {
      if (this.mode !== 'MOUSE') return;
      this.mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
      this.mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;
      this.updateMouseTarget();
    });

    window.addEventListener('mousedown', (e) => {
      if (this.mode !== 'MOUSE') return;
      if (e.button === 0) {
        this.currentAction = ActionState.PINCH_FEED;
        this.feedTriggered = true; // 点击瞬间只触发一次吐舌
      }
    });

    window.addEventListener('mouseup', () => {
      if (this.mode !== 'MOUSE') return;
      this.currentAction = ActionState.FOLLOW;
    });
  }

  updateMouseTarget() {
    this.raycaster.setFromCamera(this.mouse, this.camera);
    this.raycaster.ray.intersectPlane(this.plane, this.targetPosition);
  }

  onHandUpdate(handData) {
    if (this.mode !== 'CAMERA') return;
    if (!handData.detected) {
      this.currentAction = ActionState.IDLE;
      return;
    }

    // 映射食指尖坐标到 3D 世界空间
    const ndc = new THREE.Vector3(
      -(handData.indexTip.x * 2 - 1), // 镜像翻转符合直觉
      -(handData.indexTip.y * 2 - 1),
      0.5
    );
    ndc.unproject(this.camera);
    const dir = ndc.sub(this.camera.position).normalize();
    const distance = -this.camera.position.z / dir.z;
    this.targetPosition.copy(this.camera.position).add(dir.multiplyScalar(distance));

    // 根据手势特征派发状态
    if (handData.pinchStarted) {
      // 捏合【瞬间】：置位一次性触发标记，由主循环消费后清零，避免捏住不放反复吐舌
      this.currentAction = ActionState.PINCH_FEED;
      this.feedTriggered = true;
    } else if (handData.isPinching) {
      // 持续捏合：保持对焦注视，但不再重复触发吐舌
      this.currentAction = ActionState.PINCH_FEED;
    } else if (handData.isOpenPalm) {
      this.currentAction = ActionState.ALERT;
    } else {
      this.currentAction = ActionState.FOLLOW;
    }
  }
}
```

## `src/input/HandTracker.js`
> MediaPipe Hands 封装：加载摄像头、运行手部关键点识别，对食指尖做 EMA 平滑滤波，用迟滞阈值判断捏合（pinch）/ 张掌（open palm），并额外输出 `pinchStarted` / `pinchReleased` 动作事件（只在状态翻转帧置位），回调给 `InputManager`。

```js
import { Hands } from '@mediapipe/hands';
import { Camera } from '@mediapipe/camera_utils';

export class HandTracker {
  constructor(onResultCallback) {
    this.onResult = onResultCallback;
    this.videoElement = document.createElement('video');
    this.videoElement.style.display = 'none';
    document.body.appendChild(this.videoElement);

    // 滤波平滑向量
    this.smoothedIndex = { x: 0.5, y: 0.5, z: 0 };
    this.isPinchingPrev = false;

    this.hands = new Hands({
      locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/hands/${file}`
    });

    this.hands.setOptions({
      maxNumHands: 1,
      modelComplexity: 1,
      minDetectionConfidence: 0.7,
      minTrackingConfidence: 0.7
    });

    this.hands.onResults((results) => this.processResults(results));
    this.cameraUtils = null;
  }

  start() {
    if (!this.cameraUtils) {
      this.cameraUtils = new Camera(this.videoElement, {
        onFrame: async () => {
          await this.hands.send({ image: this.videoElement });
        },
        width: 640,
        height: 480
      });
    }
    this.cameraUtils.start();
  }

  stop() {
    if (this.cameraUtils) this.cameraUtils.stop();
  }

  processResults(results) {
    if (!results.multiHandLandmarks || results.multiHandLandmarks.length === 0) {
      this.onResult({ detected: false });
      return;
    }

    const lm = results.multiHandLandmarks[0];
    const thumbTip = lm[4];
    const rawIndex = lm[8];
    const middleTip = lm[12];
    const wrist = lm[0];

    // 指数移动平均平滑滤波 (EMA Filter, Alpha = 0.28) 彻底滤除微颤
    const alpha = 0.28;
    this.smoothedIndex.x += (rawIndex.x - this.smoothedIndex.x) * alpha;
    this.smoothedIndex.y += (rawIndex.y - this.smoothedIndex.y) * alpha;
    this.smoothedIndex.z += (rawIndex.z - this.smoothedIndex.z) * alpha;

    // 捏合迟滞检测 (Hysteresis - 进入阈值 0.06，释放阈值 0.085)
    const pinchDist = Math.hypot(thumbTip.x - rawIndex.x, thumbTip.y - rawIndex.y, thumbTip.z - rawIndex.z);
    let isPinching = this.isPinchingPrev;
    if (pinchDist < 0.06) isPinching = true;
    else if (pinchDist > 0.085) isPinching = false;
    // 动作化手势：只在状态翻转的那一帧置位，避免"一直捏着"被当成持续触发
    const pinchStarted = isPinching && !this.isPinchingPrev;
    const pinchReleased = !isPinching && this.isPinchingPrev;
    this.isPinchingPrev = isPinching;

    // 张开手掌检测
    const palmSpan = Math.hypot(middleTip.x - wrist.x, middleTip.y - wrist.y);
    const isOpenPalm = palmSpan > 0.46 && !isPinching;

    this.onResult({
      detected: true,
      indexTip: this.smoothedIndex,
      isPinching: isPinching,
      isOpenPalm: isOpenPalm,
      pinchStarted: pinchStarted,
      pinchReleased: pinchReleased
    });
  }
}
```
